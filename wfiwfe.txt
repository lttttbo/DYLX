cpu_vcpu_emulate_wfi_wfe.001.R.condition_false_next_16bit
Begin User Code
static struct vmm_vcpu vcpu;
static arch_regs_t regs;
static u32 iss;

memset(&vcpu, 0, sizeof(vcpu));
memset(&regs, 0, sizeof(regs));

/*
 * 显式构造 ISS：
 * ISS_CV_MASK 有效，cond = 0x0，即 EQ。
 */
iss = 0U;
iss |= (u32)ISS_CV_MASK;
iss |= (u32)(((u32)0x0U << (u32)ISS_COND_SHIFT) & (u32)ISS_COND_MASK);

/*
 * Z = 0，所以 EQ 条件不成立。
 * cpu_vcpu_condition_check 返回 false。
 */
regs.pstate = 0U;
regs.pc = 0x1000ULL;

<<cpu_vcpu_helper.cpu_vcpu_emulate_wfi_wfe.vcpu>> = &vcpu;
<<cpu_vcpu_helper.cpu_vcpu_emulate_wfi_wfe.regs>> = &regs;
<<cpu_vcpu_helper.cpu_vcpu_emulate_wfi_wfe.il>> = 0U;
<<cpu_vcpu_helper.cpu_vcpu_emulate_wfi_wfe.iss>> = iss;
期望值
return = 0
regs.pc = 0x1002

vcast_wfi_ready_count_call = 0
vcast_wfi_yield_call = 0
vcast_wfi_wait_call = 0
vcast_wfi_update_itstate_call = 0
注释
测试目的：验证指令条件不满足时，函数不执行 WFI/WFE 行为，仅推进到下一条 16-bit 指令。 用例类别：基于需求的测试 预置条件：构造有效 vcpu 和 regs；显式构造 iss，使条件码为 EQ；设置 Z 标志未置位；设置 il=0。 测试步骤： step1、清零 vcpu 和 regs。 step2、构造 iss，设置 ISS_CV_MASK 和条件码 0x0。 step3、设置 regs.pstate=0，使 EQ 条件不成立。 step4、设置 regs.pc=0x1000。 step5、设置 il=0。 step6、调用 cpu_vcpu_emulate_wfi_wfe。 step7、观察返回值、regs.pc 和各外部依赖调用次数。 用例来源：需求分析。 追踪ID：cpu_vcpu_emulate_wfi_wfe。 等价列表：条件不满足，16-bit 指令长度。 其它说明：覆盖 cpu_vcpu_condition_check(...) 返回 false、il ? 4 : 2 中 il=false、Thumb ITSTATE 更新条件为 false 的路径。 通过准则：函数返回 VMM_OK=0；regs.pc=0x1002；ready count、yield、wait 和 update itstate 均未调用。

static arch_regs_t regs;
static u32 iss;

memset(&vcpu, 0, sizeof(vcpu));
memset(&regs, 0, sizeof(regs));

四、配置差异说明
如果当前工程定义了：
CONFIG_WFE_WFI_IDLE_MODE
则下面代码不会参与编译：
vcpu_ready_count = vmm_scheduler_current_ready_vcpu_count();
if (vcpu_ready_count <= 1) {
    return VMM_OK;
}
这种情况下：
cpu_vcpu_emulate_wfi_wfe.002.R.ready_count_le_1_early_return
对应分支属于配置裁剪，不应要求覆盖。其余 3 条仍然适用。
