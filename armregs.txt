board_id

后续再继续：
arm_sysregs_reg_read
arm_sysregs_reg_write
arm_sysregs_emulator_read8/read16/read32

arm_sysregs_emulator_reset

arm_sysregs_emulator_init/exit
 
⸻
 
一、User Code / Stub 方案
1. Test Case Header User Code
改写目的： 让每条用例的 Begin User Code 中可以使用 memset。
#include <string.h>
 
⸻
 
2. User Global / Data
改写目的： 为外部副作用函数和时间函数提供可控返回值；只定义简单变量，不定义新的辅助函数，减少额外测试负担。
unsigned long long vcast_sysregs_timestamp;

int vcast_sysregs_shutdown_rc;
int vcast_sysregs_reboot_rc;
int vcast_sysregs_irq_rc;

unsigned int vcast_sysregs_last_irq;
unsigned int vcast_sysregs_last_irq_level;
unsigned int vcast_sysregs_last_irq_arg;
 
⸻
 
3. Stub：
vmm_timer_timestamp
改写目的： arm_sysregs_reg_read 和 arm_sysregs_emulator_reset 依赖时间戳。单元测试中需要固定时间，避免非确定性结果。
return vcast_sysregs_timestamp;
 
⸻
 
4. Stub：
vmm_manager_guest_shutdown_request
改写目的： 隔离真实 guest shutdown 副作用。
return vcast_sysregs_shutdown_rc;
 
⸻
 
5. Stub：
vmm_manager_guest_reboot_request
改写目的： 隔离真实 guest reboot 副作用。
return vcast_sysregs_reboot_rc;
 
⸻
 
6. Stub：
vmm_devemu_emulate_irq
改写目的： arm_sysregs_reg_write 的 PB926 CLCD 分支会触发虚拟中断。这里记录入参并返回可控值，避免真实设备模拟副作用。
如果你的 stub 参数名是 guest, irq, level，写：
vcast_sysregs_last_irq = irq;
vcast_sysregs_last_irq_level = level;
vcast_sysregs_last_irq_arg = 0U;

return vcast_sysregs_irq_rc;
如果 VectorCAST 生成的参数名不同，按实际参数名替换即可。
 
⸻
 
7. Stub：
vmm_printf
改写目的： 避免错误路径中的打印格式化副作用，使未实现路径可稳定运行。
return 0;
 
⸻
 
8. Lock 函数处理
如果 VectorCAST 对下面函数生成 stub，建议设为空实现：
vmm_read_lock
vmm_read_unlock
vmm_write_lock
vmm_write_unlock
改写目的： 被测逻辑不依赖锁的运行效果，单元测试中避免真实锁实现带来的环境依赖。
示例：
return;
如果这些是宏或 inline，且当前编译运行无问题，可以不处理。
 
⸻
 
9. Test Case Init 默认值
每条测试用例默认初始化：
vcast_sysregs_timestamp = 0ULL;

vcast_sysregs_shutdown_rc = 0;
vcast_sysregs_reboot_rc = 0;
vcast_sysregs_irq_rc = 0;

vcast_sysregs_last_irq = 0U;
vcast_sysregs_last_irq_level = 0U;
vcast_sysregs_last_irq_arg = 0U;
 
⸻
 
二、公共对象构造模板
涉及 struct arm_sysregs 的用例，在 Test Case User Code -> Begin 中先写：
static struct arm_sysregs s;
static struct vmm_guest guest;

memset(&s, 0, sizeof(s));
memset(&guest, 0, sizeof(guest));

s.guest = &guest;
涉及 vexpress_cfgctrl_read/write 的用例，需要额外准备 daughterboard 数组：
static unsigned int db_voltage[4];
static unsigned int db_clock[4];
static unsigned int val;

memset(db_voltage, 0, sizeof(db_voltage));
memset(db_clock, 0, sizeof(db_clock));
val = 0U;

s.db_voltage = db_voltage;
s.db_clock = db_clock;

s.db_num_vsensors = 2U;
s.db_num_clocks = 3U;

db_voltage[0] = 1000000U;
db_voltage[1] = 1100000U;

db_clock[0] = 45000000U;
db_clock[1] = 23750000U;
db_clock[2] = 66670000U;

s.mb_clock[0] = 50000000U;
s.mb_clock[1] = 23750000U;
s.mb_clock[2] = 24000000U;
s.mb_clock[3] = 24000000U;
s.mb_clock[4] = 24000000U;
s.mb_clock[5] = 24000000U;
说明： vexpress_cfgctrl_read/write 中判断 motherboard clock 时使用的是：
device < sizeof(s->mb_clock)
这在当前源码中是按字节数判断，sizeof(s->mb_clock)=24。用例里访问 motherboard clock 时只使用 device=0..5，避免数组越界。
