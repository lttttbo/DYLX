对，之前那版 helper 确实偏多。从覆盖实现的角度，不需要那么多 get 方法。更简洁的做法是：
	•	Unit Appendix 里定义真实对象和少量初始化函数；
	•	通过少数几个全局指针直接在 Value User Code 中赋值；
	•	只保留无法在 Value User Code 中直接完成的初始化 helper，例如初始化内部 struct vmm_semaphore_resource、设置 sem->value。

下面是精简版。

⸻

/*
 * VectorCAST semaphore fixture.
 * Keep object definitions here because struct vmm_semaphore_resource
 * is private to vmm_semaphore.c.
 */

static struct vmm_semaphore vcast_sem_obj;
static struct vmm_semaphore_resource vcast_sem_res1_obj;
static struct vmm_semaphore_resource vcast_sem_res2_obj;

static unsigned long vcast_sem_cur_vcpu_storage;
static unsigned long vcast_sem_other_vcpu_storage;

/* Global pointer handles used directly by Value User Code */
struct vmm_semaphore *vcast_sem = &vcast_sem_obj;
struct vmm_vcpu *vcast_sem_cur_vcpu =
    (struct vmm_vcpu *)&vcast_sem_cur_vcpu_storage;
struct vmm_vcpu *vcast_sem_other_vcpu =
    (struct vmm_vcpu *)&vcast_sem_other_vcpu_storage;
struct vmm_vcpu_resource *vcast_sem_res1_vcpu_res =
    &vcast_sem_res1_obj.res;
void *vcast_sem_zalloc_res = (void *)&vcast_sem_res1_obj;

static void vcast_sem_zero(void *ptr, unsigned long size)
{
    unsigned char *p = (unsigned char *)ptr;

    while (size > 0UL) {
        *p = 0U;
        ++p;
        --size;
    }
}

void vcast_sem_reset(unsigned int value, unsigned int limit)
{
    vcast_sem_zero(&vcast_sem_obj, sizeof(vcast_sem_obj));
    vcast_sem_zero(&vcast_sem_res1_obj, sizeof(vcast_sem_res1_obj));
    vcast_sem_zero(&vcast_sem_res2_obj, sizeof(vcast_sem_res2_obj));

    INIT_SPIN_LOCK(&vcast_sem_obj.wq.lock);
    INIT_LIST_HEAD(&vcast_sem_obj.wq.vcpu_list);
    INIT_LIST_HEAD(&vcast_sem_obj.res_list);

    vcast_sem_obj.wq.vcpu_count = 0U;
    vcast_sem_obj.value = value;
    vcast_sem_obj.limit = limit;
}

void vcast_sem_force_value(unsigned int value)
{
    vcast_sem_obj.value = value;
}

/*
 * slot:
 *   1 -> vcast_sem_res1_obj
 *   2 -> vcast_sem_res2_obj
 *
 * owner:
 *   0 -> vcast_sem_cur_vcpu
 *   1 -> vcast_sem_other_vcpu
 *
 * sem_valid:
 *   0 -> resource.sem = NULL
 *   1 -> resource.sem = vcast_sem
 */
void vcast_sem_add_resource(unsigned int slot,
                            unsigned int owner,
                            unsigned int count,
                            unsigned int sem_valid)
{
    struct vmm_semaphore_resource *res;

    if (slot == 2U) {
        res = &vcast_sem_res2_obj;
    } else {
        res = &vcast_sem_res1_obj;
    }

    vcast_sem_zero(res, sizeof(*res));

    INIT_LIST_HEAD(&res->head);

    res->count = count;
    res->sem = sem_valid ? &vcast_sem_obj : 0;
    res->vcpu = owner ? vcast_sem_other_vcpu : vcast_sem_cur_vcpu;
    res->res.name = "vmm_semaphore";
    res->res.cleanup = __vmm_semaphore_cleanup;

    if (sem_valid) {
        list_add_tail(&res->head, &vcast_sem_obj.res_list);
    }
}

十一、不建议覆盖的分支说明

sem == NULL
sem->limit == 0
vmm_zalloc 返回 NULL
__vmm_semaphore_cleanup 中的 !sres
__vmm_semaphore_cleanup 中 vcpu_res == NULL

这些不建议作为普通单元测试执行：
	•	sem == NULL 和 sem->limit == 0 由 BUG_ON 处理，不是普通错误码路径；
	•	vmm_zalloc 返回 NULL 后会触发 BUG_ON(!sres)；
	•	__vmm_semaphore_cleanup 中 sres 是 container_of(vcpu_res, ...) 计算得到，而且在 if (!sres ...) 之前已经访问了 sres->sem，因此 !sres 分支不适合通过普通测试覆盖；
	•	vcpu_res == NULL 会先参与 container_of，存在非法地址风险。

覆盖率说明可写：

该分支属于断言或不可安全构造路径。当前实现会在普通返回判断前进入 BUG_ON 或进行 container_of 指针运算，无法通过普通单元测试安全覆盖。相关正常路径、错误传播路径和可构造的无效等价类已通过其它用例覆盖。
