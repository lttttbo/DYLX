## Xvisor 支持 OP-TEE 虚拟化 - 技术方案

### 1. 背景概述

**OP-TEE 虚拟化架构**：OP-TEE 具有实验性的虚拟化支持，允许单个 OP-TEE 实例为多个虚拟机运行可信应用 (TA)。其核心思想是隔离所有 VM 相关状态，使 VM 之间互不影响。

**核心挑战**：
- Hypervisor 是唯一知道哪个 VM 正在调用 OP-TEE 的实体
- VM 使用中间物理地址 (IPA)，而 OP-TEE 需要真实物理地址 (PA)
- 需要一个 "TEE Mediator" 组件来处理协议转换和地址翻译

目前只有 **XEN** hypervisor 实现了 OP-TEE mediator，你需要为 Xvisor 实现类似组件。

---

### 2. 实现架构

```
+------------------+     +------------------+
|      VM 1        |     |      VM 2        |
|  (Normal World)  |     |  (Normal World)  |

+--------------------------------------------------+
```

### 3. TEE Mediator 核心功能实现

#### 3.1 SMC 调用处理 (必须实现的 SMC)

```c
// 文件建议路径: arch/arm/cpu/arm64/tee/optee_mediator.c

#include <vmm_types.h>
#include <vmm_error.h>
#include <vmm_vcpu_irq.h>
#include <cpu_vcpu_smc.h>

/* OP-TEE SMC Function IDs */
#define OPTEE_SMC_VM_CREATED       0x32000000
#define OPTEE_SMC_VM_DESTROYED     0x32000001
#define OPTEE_SMC_STD_CALL_MASK    0xFF000000
#define OPTEE_SMC_STD_CALL_VAL     0x32000000

/* 保留的 Hypervisor 客户端 ID */
#define HYP_CLNT_ID                0

struct optee_vm_state {
    u32 vm_id;
    bool created;
    /* 共享内存追踪 */
    struct list_head shm_list;
    spinlock_t lock;
};

/* VM 创建通知 OP-TEE */
int optee_vm_create(struct vmm_guest *guest)
{
    struct arm_smccc_res res;
    u32 vm_id = guest->id + 1;  /* VM ID 0 保留给 hypervisor */
    
    /* 调用 OPTEE_SMC_VM_CREATED */
    arm_smccc_smc(OPTEE_SMC_VM_CREATED, 
                  vm_id,        /* a1: VM ID */
                  0, 0, 0, 0, 0,
                  HYP_CLNT_ID,  /* a7: 调用者是 hypervisor */
                  &res);
    
    if (res.a0 != OPTEE_SMC_RETURN_OK) {
        return VMM_EFAIL;
    }
    
    return VMM_OK;
}

/* VM 销毁通知 OP-TEE */
int optee_vm_destroy(struct vmm_guest *guest)
{
    struct arm_smccc_res res;
    u32 vm_id = guest->id + 1;
    
    /* 首先停止所有 VCPU，确保 OP-TEE 没有活动线程 */
    vmm_guest_halt_vcpus(guest);
    
    /* 调用 OPTEE_SMC_VM_DESTROYED */
    arm_smccc_smc(OPTEE_SMC_VM_DESTROYED,
                  vm_id,
                  0, 0, 0, 0, 0,
                  HYP_CLNT_ID,
                  &res);
    
    return VMM_OK;
}
```

#### 3.2 SMC 拦截与转发

```c
/* SMC 处理主函数 - 需要集成到 Xvisor 的 SMC trap 处理路径 */
int optee_handle_smc(struct vmm_vcpu *vcpu, 
                     arch_regs_t *regs,
                     u32 smc_fid)
{
    struct arm_smccc_res res;
    physical_addr_t pa;
    u32 vm_id = vcpu->guest->id + 1;
    
    /* 检查是否是 OP-TEE 相关的 SMC */
    if (!is_optee_smc(smc_fid)) {
        return VMM_ENOTSUPP;  /* 让其他处理器处理 */
    }
    
    /* IPA -> PA 地址转换 (针对 a1-a6 中的地址参数) */
    /* 这里需要根据具体 SMC 类型决定哪些参数是地址 */
    if (smc_needs_addr_translation(smc_fid)) {
        /* 转换 a1 中的地址 (如果是地址参数) */
        if (is_addr_param(smc_fid, 1)) {
            if (vmm_guest_ipa2pa(vcpu->guest, 
                                 regs->a1, 
                                 &pa) != VMM_OK) {
                regs->a0 = OPTEE_SMC_RETURN_EBADADDR;
                return VMM_OK;
            }
            regs->a1 = pa;
        }
        /* 类似处理 a2-a6 */
    }
    
    /* Pin 共享内存页面 */
    if (smc_shares_memory(smc_fid)) {
        optee_pin_shared_mem(vcpu->guest, regs);
    }
    
    /* 注入 VM ID 到 a7 并转发 SMC */
    arm_smccc_smc(smc_fid,
                  regs->a1, regs->a2, regs->a3,
                  regs->a4, regs->a5, regs->a6,
                  vm_id,    /* a7: VM ID */
                  &res);
    
    /* 返回结果给 Guest */
    regs->a0 = res.a0;
    regs->a1 = res.a1;
    regs->a2 = res.a2;
    regs->a3 = res.a3;
    
    return VMM_OK;
}
```

#### 3.3 IPA 到 PA 地址转换

```c
/* 地址转换函数 */
static int translate_ipa_to_pa(struct vmm_guest *guest,
                               physical_addr_t ipa,
                               physical_addr_t *pa)
{
    int rc;
    struct vmm_region *reg;
    
    /* 使用 Xvisor 的 guest 地址转换 API */
    rc = vmm_guest_physical_map(guest, ipa, 
                                PAGE_SIZE, pa, 
                                NULL, NULL);
    if (rc) {
        return VMM_EFAIL;
    }
    
    return VMM_OK;
}

/* 处理 RPC 命令缓冲区中的地址 */
static int translate_rpc_buffer(struct vmm_guest *guest,
                                physical_addr_t buf_pa,
                                size_t buf_size)
{
    struct optee_msg_arg *arg;
    int i;
    
    /* 映射缓冲区 */
    arg = vmm_host_memmap(buf_pa, buf_size, VMM_MEMORY_READWRITE);
    if (!arg) {
        return VMM_EFAIL;
    }
    
    /* 转换参数中的所有地址 */
    for (i = 0; i < arg->num_params; i++) {
        if (arg->params[i].attr == OPTEE_MSG_ATTR_TYPE_TMEM_INPUT ||
            arg->params[i].attr == OPTEE_MSG_ATTR_TYPE_TMEM_OUTPUT ||
            arg->params[i].attr == OPTEE_MSG_ATTR_TYPE_TMEM_INOUT) {
            
            physical_addr_t pa;
            if (translate_ipa_to_pa(guest, 
                                    arg->params[i].u.tmem.buf_ptr,
                                    &pa) == VMM_OK) {
                arg->params[i].u.tmem.buf_ptr = pa;
            }
        }
    }
    
    vmm_host_memunmap(arg);
    return VMM_OK;
}
```

#### 3.4 共享内存页面 Pin 机制

```c
struct optee_shm_entry {
    struct list_head list;
    physical_addr_t guest_ipa;
    physical_addr_t host_pa;
    size_t size;
    int pin_count;
};

/* Pin 共享内存页面 */
static int optee_pin_shared_mem(struct vmm_guest *guest,
                                physical_addr_t ipa,
                                size_t size)
{
    struct optee_vm_state *state = guest->tee_priv;
    struct optee_shm_entry *entry;
    physical_addr_t pa;
    
    /* 转换地址 */
    if (translate_ipa_to_pa(guest, ipa, &pa) != VMM_OK) {
        return VMM_EFAIL;
    }
    
    /* 创建追踪条目 */
    entry = vmm_zalloc(sizeof(*entry));
    entry->guest_ipa = ipa;
    entry->host_pa = pa;
    entry->size = size;
    entry->pin_count = 1;
    
    /* 标记页面为 pinned，防止被交换或重映射 */
    vmm_guest_pin_region(guest, ipa, size);
    
    spin_lock(&state->lock);
    list_add(&entry->list, &state->shm_list);
    spin_unlock(&state->lock);
    
    return VMM_OK;
}

/* Unpin 共享内存 */
static int optee_unpin_shared_mem(struct vmm_guest *guest,
                                  physical_addr_t ipa)
{
    struct optee_vm_state *state = guest->tee_priv;
    struct optee_shm_entry *entry;
    
    spin_lock(&state->lock);
    list_for_each_entry(entry, &state->shm_list, list) {
        if (entry->guest_ipa == ipa) {
            entry->pin_count--;
            if (entry->pin_count == 0) {
                list_del(&entry->list);
                vmm_guest_unpin_region(guest, ipa, entry->size);
                vmm_free(entry);
            }
            break;
        }
    }
    spin_unlock(&state->lock);
    
    return VMM_OK;
}
```

### 4. Xvisor 集成点

需要修改 Xvisor 以下位置来集成 OP-TEE mediator：

| 文件/模块 | 修改内容 |
|----------|---------|
| `arch/arm/cpu/arm64/cpu_vcpu_helper.c` | 添加 SMC trap 到 mediator 的分发 |
| `core/vmm_guest.c` | VM 生命周期钩子 (create/destroy) |
| `arch/arm/cpu/arm64/include/cpu_vcpu_smc.h` | SMC 处理接口定义 |
| `Kconfig` | 添加 `CONFIG_ARM_OPTEE_MEDIATOR` 选项 |

### 5. 配置与构建

#### Xvisor 配置

```kconfig
# arch/arm/Kconfig 添加
config ARM_OPTEE_MEDIATOR
    bool "OP-TEE virtualization mediator"
    depends on ARM64
    default n
    help
      Enable OP-TEE mediator support for running OP-TEE
      with multiple virtual machines.
```

#### OP-TEE 配置

```makefile
# OP-TEE 编译参数
CFG_VIRTUALIZATION=y
```

### 7. 开发建议

1. **从 QEMU-V8 开始测试** - 这是唯一被官方测试过的平台

2. **分阶段实现**：
   - 阶段1: VM 生命周期管理 (VM_CREATED/VM_DESTROYED)
   - 阶段2: 基本 SMC 转发与 VM ID 注入
   - 阶段3: IPA->PA 地址转换
   - 阶段4: 共享内存 Pin 机制
   - 阶段5: 完整 RPC 缓冲区处理

3. **注意事项**：
   - OP-TEE 虚拟化仅支持 ARMv8
   - VM 数量和内存是静态分配的
   - 启用虚拟化后 OP-TEE 无法在非虚拟化环境运行
