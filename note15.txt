下面给一版 emulate_arm.c 的单元测试设计基线。这个文件函数非常多，直接逐条硬编码会非常难维护，所以建议采用：
	1.	静态函数直接测：arm_inst_ldrh_i、arm_inst_str_i 等 static 函数都单独作为 UUT 测；
	2.	分组函数用 SBF 隔离派发逻辑：arm_instgrp_dataproc、arm_instgrp_ldrstr、arm_instgrp_coproc 等主要验证派发分支，不重复验证子函数内部；
	3.	少量 User Code 变量 + 通用 Stub：寄存器读写、内存读写、coproc 回调都用全局变量控制，避免每条用例重复写大量代码；
	4.	无法到达的 dead code 明确排除。

/*
 * 用于构造 instruction bit field。
 * 注意：只用于 32 bit 指令字段，避免 1U << 32。
 */
#define VCAST_ARM_FIELD_MASK(end, start) \
    ((u32)((((u64)1ULL << ((end) - (start) + 1U)) - 1ULL) << (start)))

#define VCAST_ARM_SET_FIELD(inst, end, start, val)             \
    do {                                                       \
        (inst) &= ~VCAST_ARM_FIELD_MASK((end), (start));       \
        (inst) |= (((u32)(val) << (start)) &                   \
                   VCAST_ARM_FIELD_MASK((end), (start)));      \
    } while (0)

#define VCAST_ARM_SET_BIT(inst, bit, val)                      \
    do {                                                       \
        if (val) {                                             \
            (inst) |= ((u32)1U << (bit));                      \
        } else {                                               \
            (inst) &= ~((u32)1U << (bit));                     \
        }                                                      \
    } while (0)

#define VCAST_ARM_SET_COND(inst, cond)                         \
    do {                                                       \
        (inst) &= ~ARM_INST_COND_MASK;                         \
        (inst) |= (((u32)(cond) << ARM_INST_COND_SHIFT) &      \
                   ARM_INST_COND_MASK);                        \
    } while (0)

#define VCAST_COND_EQ     0x0U
#define VCAST_COND_NE     0x1U
#define VCAST_COND_AL     0xEU
#define VCAST_COND_NV     0xFU

#define VCAST_ARM_INIT_REGS(regs_ptr, pc_value, cpsr_value)    \
    do {                                                       \
        memset((regs_ptr), 0, sizeof(*(regs_ptr)));            \
        arm_pc((regs_ptr)) = (pc_value);                       \
        arm_cpsr((regs_ptr)) = (cpsr_value);                   \
    } while (0)

#define VCAST_ARM_RESET_ARRAYS()                               \
    do {                                                       \
        memset(vcast_reg_read_value, 0, sizeof(vcast_reg_read_value)); \
        memset(vcast_reg_write_value, 0, sizeof(vcast_reg_write_value)); \
        memset(vcast_reg_write_called, 0, sizeof(vcast_reg_write_called)); \
        memset(vcast_mem_read_value_seq, 0, sizeof(vcast_mem_read_value_seq)); \
        memset(vcast_mem_read_rc_seq, 0, sizeof(vcast_mem_read_rc_seq)); \
        memset(vcast_mem_write_rc_seq, 0, sizeof(vcast_mem_write_rc_seq)); \
        memset(vcast_mem_write_addr_seq, 0, sizeof(vcast_mem_write_addr_seq)); \
        memset(vcast_mem_write_size_seq, 0, sizeof(vcast_mem_write_size_seq)); \
        memset(vcast_mem_write_value_seq, 0, sizeof(vcast_mem_write_value_seq)); \
        vcast_mem_read_call = 0U;                              \
        vcast_mem_write_call = 0U;                             \
        vcast_irq_assert_called = 0U;                          \
        vcast_irq_assert_irq = 0U;                             \
        vcast_cpu_halt_called = 0U;                            \
    } while (0)

这些函数是为了覆盖 stc/ldc/mcr/mrc/cdp 的成功路径。函数指针签名不同，无法完全避免新增函数。

                                   arch_regs_t *regs,

                                   bool uopt, u32 imm8)
{
    (void)vcpu; (void)regs; (void)D; (void)CRd; (void)uopt; (void)imm8;

op != 0:
  op1=10000 -> movw_i
  op1=10100 -> fallthrough unpredictable
  op1=10010/10110 -> fallthrough unpredictable
  default -> fallthrough unpredictable

op == 0:
  unprivileged extra load/store:
    op2=1011 op1 bit0=1 -> ldrht
    op2=1011 op1 bit0=0 -> strht
    op2=1101 bit0=1 -> ldrsbt
    op2=1111 bit0=1 -> ldrsht

  extra load/store:
    op2=1011 -> strh_r / ldrh_r / strh_i / ldrh_l / ldrh_i
    op2=1101 -> ldrd_r / ldrsb_r / ldrd_l / ldrd_i / ldrsb_l / ldrsb_i

    op1=11000 -> strex
    op1=11001 -> ldrex

  no match -> unpredictable

十四、明确 dead code / 覆盖率排除
1. 
arm_inst_movw_i
源码：
if (Rd == 15) {
    arm_unpredictable(...);
    return VMM_EFAIL;
}
...
if (arm_condition_passed(cond, regs)) {
    if (Rd == 15) {
        arm_pc(regs) = result;
    } else {
        cpu_vcpu_reg_write(...);
    }
}
内部：
if (Rd == 15) {
    arm_pc(regs) = result;
}
不可达，因为前面 Rd == 15 已经 return。
 
排除说明：
`arm_inst_movw_i` 中 condition passed 内部的 `if (Rd == 15)` 分支不可达。函数前部已经对 `Rd == 15` 执行不可预测处理并返回，后续相同条件不可能再次成立，属于源代码逻辑 dead code，应从覆盖率排除。
 
⸻
 
2. 
emulate_arm_inst
switch default
switch (op1 & 0x6) {
case 0x0:
case 0x2:
case 0x4:
case 0x6:
...
default:
    break;
}
op1 & 0x6 只能为 0/2/4/6，default 不可达。
 
排除说明：
`emulate_arm_inst` 中 `switch (op1 & 0x6)` 的 default 分支不可达，因为按位与 `0x6` 的结果只能为 0x0、0x2、0x4、0x6。该 default 属于防御性 dead code，应从覆盖率排除。
 
⸻

十五、建议的最小高覆盖用例清单
完整文件建议分批执行，避免一次过多难以调试：
基础函数：
arm_unpredictable.001.R
arm_sign_extend.001.R
arm_sign_extend.002.R
arm_condition_check.001.R ~ .016.R
arm_decode_imm_shift.001.R ~ .008.IN.type.NEC
arm_shift_c.001.R ~ .008.R
arm_add_with_carry.001.R ~ .004.R

单指令 handler：
每个 load/store handler 按对应矩阵覆盖 normal / condition false / unpredictable / mem fail / wback。
exclusive 指令按 ldrex/strex 矩阵。
block transfer 按 ldm/stm 矩阵。
coproc handler 按 stcx/ldcx/mcr/mrc/cdp 矩阵。

分组 dispatcher：
arm_instgrp_dataproc 派发矩阵
arm_instgrp_ldrstr 派发矩阵
arm_instgrp_brblk 派发矩阵
arm_instgrp_coproc 派发矩阵
arm_instgrp_uncond.001.R ~ .003.IN.op1.NEC
emulate_arm_inst.001.IN.vcpu.NEC ~ dispatch 全路径
这套设计能保证：
if 判断两侧都覆盖；
switch case 均覆盖；
unpredictable/error/condition false/normal path 均覆盖；
内存读写失败和成功均覆盖；
写回和非写回均覆盖；
子分组 dispatcher 派发全部覆盖；
dead code 单独说明并排除。
