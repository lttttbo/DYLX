下面这组是：
arm_inst_ldrsh_r
arm_inst_ldrsb_r
也就是 signed halfword / signed byte 的 register offset load。
这组字段和 ldrh_r 基本一致：
cond   bits[31:28]
P      bit[24]
U      bit[23]
W      bit[21]
Rn     bits[19:16]
Rt     bits[15:12]
Rm     bits[3:0]
核心差异：
arm_inst_ldrsh_r：读取 2 字节，执行 16 位符号扩展
arm_inst_ldrsb_r：读取 1 字节，执行 8 位符号扩展
每条用例先使用公共模板：
static struct vmm_vcpu vcpu;
static arch_regs_t regs;
unsigned int inst;

memset(&vcpu, 0, sizeof(vcpu));
memset(&regs, 0, sizeof(regs));

vcpu.is_normal = 1U;
regs.pc = 0x1000U;
regs.pstate = 0U;

inst = 0U;

add=true
wback=false
condition=true

/* cond = AL */
inst |= (0xEU << ARM_INST_COND_SHIFT);

inst |= (1U << ARM_INST_LDRSTR_P_START);
inst |= (1U << ARM_INST_LDRSTR_U_START);

/* Rn=1, Rt=2, Rm=3 */
inst |= (1U << ARM_INST_LDRSTR_RN_START);
inst |= (2U << ARM_INST_LDRSTR_RT_START);
inst |= (3U << ARM_INST_LDRSTR_RM_START);

vcast_reg_read_value[1] = 0x2000U;  /* Rn base */
vcast_reg_read_value[3] = 0x10U;    /* Rm offset */

vcast_mem_read_value_seq[0] = 0x8001U;
vcast_mem_read_rc_seq[0] = 0;
