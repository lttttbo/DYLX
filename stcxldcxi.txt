当前仍是 emulate_arm.c。这两个函数控制流基本对称，可以共用一套最精简的协处理器 User Code：
arm_inst_stcx：9 条
arm_inst_ldcx_i：9 条
不增加调用次数打点，不验证中间回调是否执行；通过返回值和 regs.pc 区分路径。下面的回调参数顺序与 cpu_coproc_ldcstc_accept/done/read/write 的函数指针定义一致。 
 
⸻
 
一、两个函数共用的最精简 User Code
已有同名内容时直接复用，不要重复定义。你之前为 arm_inst_ldcx_l 建立的这套 User Code 也可以继续使用。
1. User Global / Data
struct cpu_vcpu_coproc vcast_cp;

/*
 * ldcstc_accept 的可控返回值：
 * 0 -> FALSE
 * 1 -> TRUE
 */
unsigned int vcast_cp_accept_ret;

/*
 * ldcstc_done 的结束下标：
 * 0 -> 第一次判断即返回 TRUE，循环不执行
 * 1 -> index=0 时返回 FALSE，循环执行一次；
 *      index=1 时返回 TRUE，退出循环
 */
unsigned int vcast_cp_done_after;
注意：
struct cpu_vcpu_coproc vcast_cp;
只放在 User Global / Data 一次。不要再放到 Test Case Init 或 Begin User Code，否则会重复定义。

ldcstc_accept
回调
**改写目的：**控制协处理器接受或拒绝 STC/LDC 指令。
static bool vcast_ldcstc_accept(struct vmm_vcpu *vcpu,
                                arch_regs_t *regs,
                                u32 D,
                                u32 CRd,
                                u32 uopt,
                                u32 imm8)
{
    (void)vcpu;
    (void)regs;
    (void)D;
    (void)CRd;
    (void)uopt;
    (void)imm8;

    return vcast_cp_accept_ret ? TRUE : FALSE;
}
ldcstc_done
回调
**改写目的：**控制传输循环执行零次或一次。本批成功和故障用例统一设置为执行一次。
static bool vcast_ldcstc_done(struct vmm_vcpu *vcpu,
                              arch_regs_t *regs,
                              u32 index,
                              u32 D,
                              u32 CRd,
                              u32 uopt,
                              u32 imm8)
{
    (void)vcpu;
    (void)regs;
    (void)D;
    (void)CRd;
    (void)uopt;
    (void)imm8;

    return (index >= vcast_cp_done_after) ? TRUE : FALSE;
}
