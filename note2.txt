1. notifier_chain_register

源码：

static int notifier_chain_register(struct vmm_notifier_block **nl,
                   struct vmm_notifier_block *n)
{
    while ((*nl) != NULL) {
        if (n->priority > (*nl)->priority)
            break;
        nl = &((*nl)->next);
    }
    n->next = *nl;
    *nl = n;
    return VMM_OK;
}

需要覆盖：
	•	空链表；
	•	新节点优先级高于头节点，插入头部；
	•	新节点优先级低于已有节点，插入尾部；
	•	新节点优先级介于两个节点之间，插入中间；
	•	if (n->priority > (*nl)->priority) 的 true / false；
	•	while ((*nl) != NULL) 的 true / false。

2. notifier_chain_cond_register

源码：

static int notifier_chain_cond_register(struct vmm_notifier_block **nl,
                    struct vmm_notifier_block *n)
{
    while ((*nl) != NULL) {
        if ((*nl) == n)
            return VMM_OK;
        if (n->priority > (*nl)->priority)
            break;
        nl = &((*nl)->next);
    }
    n->next = *nl;
    *nl = n;
    return VMM_OK;
}

需要覆盖：
	•	空链表插入；
	•	已存在于头节点，直接返回；
	•	已存在于后续节点，遍历后直接返回；
	•	优先级高于当前节点，break 插入；
	•	优先级低于已有节点，尾部插入；
	•	中间插入。

3. notifier_chain_unregister

源码：

static int notifier_chain_unregister(struct vmm_notifier_block **nl,
                     struct vmm_notifier_block *n)
{
    while ((*nl) != NULL) {
        if ((*nl) == n) {
            *nl = n->next;
            return VMM_OK;
        }
        nl = &((*nl)->next);
    }
    return VMM_ENOENT;
}

需要覆盖：
	•	删除头节点；
	•	删除非头节点；

4. notifier_call_chain

源码：

static int notifier_call_chain(struct vmm_notifier_block **nl,
                unsigned long val, void *v,
                int nr_to_call, int *nr_calls)
{
    int ret = NOTIFY_DONE;
    struct vmm_notifier_block *nb, *next_nb;
    nb = *nl;
    while (nb && nr_to_call) {
        next_nb = nb->next;
        ret = nb->notifier_call(nb, val, v);
        if (nr_calls)
            (*nr_calls)++;
        if ((ret & NOTIFY_STOP_MASK) == NOTIFY_STOP_MASK)
            break;
        nb = next_nb;
        nr_to_call--;
    }
    return ret;
}

需要覆盖：
	•	空链表；
	•	nr_to_call == 0；
	•	全量调用多个节点；
	•	nr_calls == NULL；
	•	stop mask 中断；
	•	nr_to_call 限制只调用一个节点。

7. 注意事项

不要构造：

nc == NULL
n == NULL
nl == NULL
notifier_call == NULL

因为当前代码没有对此做防御。它们会导致空指针访问，不是可控错误返回路径。

如果要测试同优先级插入顺序，可以额外增加一条：

notifier_chain_register.005.R

构造已有节点和新节点优先级相同，期望新节点追加到已有同优先级节点之后。当前源码判断条件是：

if (n->priority > (*nl)->priority)

所以同优先级不会插到已有节点前面。
