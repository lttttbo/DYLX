下面继续这批函数。测试设计保持现有约定：
	•	用例类别仅使用：基于需求的测试、接口测试、故障注入测试；
	•	每条用例提供独立完整注释；
	•	正式名称保留 .R、.IN、.FI 后缀；
	•	非本 .c 文件中的函数由 VectorCAST Stub 控制；
	•	同一文件内的函数原则上按实际实现执行；
	•	输出参数通过 Stub User Code 写回；
	•	以最少用例完成语句、分支及复合条件覆盖。

 * 控制当前物理 CPU。
 */

#undef vmm_smp_processor_id
#endif

#define vmm_smp_processor_id() \
    ((u32)vcast_smp_processor_id_ret)

/*
 * 控制 Boot CPU。
 */
#ifdef vmm_smp_bootcpu_id
#undef vmm_smp_bootcpu_id
#endif

#define vmm_smp_bootcpu_id() \
    ((u32)vcast_smp_bootcpu_id_ret)

/*
 * 隔离真实 present CPU 掩码。
 *
 * vcast_present_cpu_count=1：
 *     仅遍历 CPU0。
 *
 * vcast_present_cpu_count=2：
 *     遍历 CPU0、CPU1。
 */
#ifdef for_each_present_cpu
#undef for_each_present_cpu
#endif

#define for_each_present_cpu(cpu)             \
    for ((cpu) = 0U;                          \
         (cpu) < vcast_present_cpu_count;     \
         ++(cpu))

/*
 * 避免 vmm_cpumask_of 展开后访问真实 per-CPU 掩码。
 */
#ifdef vmm_cpumask_of
#undef vmm_cpumask_of
#endif

#define vmm_cpumask_of(cpu)                   \
    ((void)(cpu),                             \
     (const struct vmm_cpumask *)             \
     &vcast_cpumask_of_store)

/*
 * hypervisor_suspend 中屏蔽系统寄存器写入。
 */
#ifdef msr_sync
#undef msr_sync
#endif

#define msr_sync(reg, val)                    \
    do {                                      \
        (void)(val);                          \
    } while (0)

1. 必须进行的测试编译隔离
save_current_registers() 和 restore_saved_registers() 会直接修改全部通用寄存器及 SP，不应在 VectorCAST 驱动上下文中真实执行，否则可能破坏驱动栈和返回地址。
建议仅在单元测试编译中增加：
#if defined(VCAST_UNIT_TEST)

#define restore_saved_registers() \
    do { } while (0)

#define save_current_registers() \
    do { } while (0)

#else

/* 保留原来的 restore_saved_registers 定义 */
/* 保留原来的 save_current_registers 定义 */

#endif
生产构建不定义 VCAST_UNIT_TEST，原实现不变。
如链接器缺少以下符号，再在 Unit Appendix 中提供：
u8 _bringup_boot_cpu;
