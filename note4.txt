/*
 * VectorCAST semaphore fixture.
 * Object definitions must be placed in Unit Appendix, not Unit Prefix.
 */

static struct vmm_semaphore vcast_sem_obj;
static struct vmm_semaphore_resource vcast_sem_res_obj;
static struct vmm_semaphore_resource vcast_sem_res2_obj;

/*
 * Use opaque storage for VCPU pointers.
 * vmm_semaphore.c only compares/passes struct vmm_vcpu pointers in these paths.
 */
static unsigned long vcast_sem_cur_vcpu_storage;
static unsigned long vcast_sem_other_vcpu_storage;

static void vcast_sem_zero(void *ptr, unsigned long size)
{
    unsigned char *p;

    p = (unsigned char *)ptr;

    while (size > 0UL) {
        *p = 0U;
        ++p;
        --size;
    }
}

static struct vmm_vcpu *vcast_sem_cur_vcpu_ptr(void)
{
    return (struct vmm_vcpu *)&vcast_sem_cur_vcpu_storage;
}

static struct vmm_vcpu *vcast_sem_other_vcpu_ptr(void)
{
    return (struct vmm_vcpu *)&vcast_sem_other_vcpu_storage;
}

void vcast_sem_init(unsigned int value, unsigned int limit)
{
    vcast_sem_zero(&vcast_sem_obj, sizeof(vcast_sem_obj));
    vcast_sem_zero(&vcast_sem_res_obj, sizeof(vcast_sem_res_obj));
    vcast_sem_zero(&vcast_sem_res2_obj, sizeof(vcast_sem_res2_obj));

    INIT_SPIN_LOCK(&vcast_sem_obj.wq.lock);
    INIT_LIST_HEAD(&vcast_sem_obj.wq.vcpu_list);
    INIT_LIST_HEAD(&vcast_sem_obj.res_list);

    vcast_sem_obj.wq.vcpu_count = 0U;
    vcast_sem_obj.value = value;
    vcast_sem_obj.limit = limit;
}

void vcast_sem_set_value(unsigned int value)
{
    vcast_sem_obj.value = value;
}

void vcast_sem_init_current_resource(unsigned int value,
                                     unsigned int limit,
                                     unsigned int res_count)
{
    vcast_sem_init(value, limit);

    INIT_LIST_HEAD(&vcast_sem_res_obj.head);

    vcast_sem_res_obj.count = res_count;
    vcast_sem_res_obj.sem = &vcast_sem_obj;
    vcast_sem_res_obj.vcpu = vcast_sem_cur_vcpu_ptr();
    vcast_sem_res_obj.res.name = "vmm_semaphore";
    vcast_sem_res_obj.res.cleanup = __vmm_semaphore_cleanup;

    list_add_tail(&vcast_sem_res_obj.head, &vcast_sem_obj.res_list);
}

void vcast_sem_init_other_resource(unsigned int value,
                                   unsigned int limit,
                                   unsigned int res_count)
{
    vcast_sem_init(value, limit);

    INIT_LIST_HEAD(&vcast_sem_res_obj.head);

    vcast_sem_res_obj.count = res_count;
    vcast_sem_res_obj.sem = &vcast_sem_obj;
    vcast_sem_res_obj.vcpu = vcast_sem_other_vcpu_ptr();
    vcast_sem_res_obj.res.name = "vmm_semaphore";
    vcast_sem_res_obj.res.cleanup = __vmm_semaphore_cleanup;

    list_add_tail(&vcast_sem_res_obj.head, &vcast_sem_obj.res_list);
}

void vcast_sem_init_two_resources(unsigned int value,
                                  unsigned int limit,
                                  unsigned int cur_count,
                                  unsigned int other_count)
{
    vcast_sem_init(value, limit);

    INIT_LIST_HEAD(&vcast_sem_res_obj.head);
    INIT_LIST_HEAD(&vcast_sem_res2_obj.head);

    vcast_sem_res_obj.count = cur_count;
    vcast_sem_res_obj.sem = &vcast_sem_obj;
    vcast_sem_res_obj.vcpu = vcast_sem_cur_vcpu_ptr();
    vcast_sem_res_obj.res.name = "vmm_semaphore";
    vcast_sem_res_obj.res.cleanup = __vmm_semaphore_cleanup;

    vcast_sem_res2_obj.count = other_count;
    vcast_sem_res2_obj.sem = &vcast_sem_obj;
    vcast_sem_res2_obj.vcpu = vcast_sem_other_vcpu_ptr();
    vcast_sem_res2_obj.res.name = "vmm_semaphore";
    vcast_sem_res2_obj.res.cleanup = __vmm_semaphore_cleanup;

    list_add_tail(&vcast_sem_res_obj.head, &vcast_sem_obj.res_list);
    list_add_tail(&vcast_sem_res2_obj.head, &vcast_sem_obj.res_list);
}

void vcast_sem_init_current_resource_no_sem(unsigned int res_count)
{
    vcast_sem_zero(&vcast_sem_res_obj, sizeof(vcast_sem_res_obj));

    INIT_LIST_HEAD(&vcast_sem_res_obj.head);

    vcast_sem_res_obj.count = res_count;
    vcast_sem_res_obj.sem = 0;
    vcast_sem_res_obj.vcpu = vcast_sem_cur_vcpu_ptr();
    vcast_sem_res_obj.res.name = "vmm_semaphore";
    vcast_sem_res_obj.res.cleanup = __vmm_semaphore_cleanup;
}

void vcast_sem_prepare_zalloc_resource(void)
{
    vcast_sem_zero(&vcast_sem_res_obj, sizeof(vcast_sem_res_obj));
}

struct vmm_semaphore *vcast_sem_get(void)
{
    return &vcast_sem_obj;
}

struct vmm_vcpu *vcast_sem_get_cur_vcpu(void)
{
    return vcast_sem_cur_vcpu_ptr();
}

struct vmm_vcpu *vcast_sem_get_other_vcpu(void)
{
    return vcast_sem_other_vcpu_ptr();
}

struct vmm_vcpu_resource *vcast_sem_get_resource(void)
{
    return &vcast_sem_res_obj.res;
}

void *vcast_sem_get_zalloc_resource(void)
{
    return (void *)&vcast_sem_res_obj;
}

unsigned int vcast_sem_get_value(void)
{
    return vcast_sem_obj.value;
}

unsigned int vcast_sem_get_limit(void)
{
    return vcast_sem_obj.limit;
}

unsigned int vcast_sem_get_current_resource_count(void)
{
    return vcast_sem_res_obj.count;
}

unsigned int vcast_sem_get_second_resource_count(void)
{
    return vcast_sem_res2_obj.count;
}

int vcast_sem_resource_list_empty(void)
{
    return list_empty(&vcast_sem_obj.res_list);
}

十一、不建议普通执行的用例说明

以下路径不建议作为普通返回值用例直接执行：

sem == NULL
sem->limit == 0
vmm_zalloc 返回 NULL
vcpu_res == NULL 传入 __vmm_semaphore_cleanup

原因：

sem == NULL 和 sem->limit == 0 由 BUG_ON 处理，不是普通错误码返回路径。
vmm_zalloc 返回 NULL 后代码会执行 BUG_ON(!sres)，同样属于断言路径。
__vmm_semaphore_cleanup 在检查 vcpu_res 之前先执行 container_of(vcpu_res, ...)，因此 vcpu_res == NULL 存在未定义指针运算风险，不建议普通运行构造。

如果项目强制要求覆盖这些断言路径，建议统一使用 BUG_ON 测试替身，记录断言触发并终止当前用例，不与普通功能用例混用。
