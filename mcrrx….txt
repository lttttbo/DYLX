一、coprocessor 函数共用的精简 User Code
如果你已经有前面 stcx / ldcx_l 的精简 coprocessor usercode，需要把它替换成下面的扩展版，不要追加重复定义。
1. User Global / Data
已存在的变量不要重复定义，缺哪个补哪个：
unsigned int vcast_cp_mask;
unsigned int vcast_cp_accept_ret;
unsigned int vcast_cp_done_after;
unsigned int vcast_cp_read_value_seq[8];

unsigned int vcast_cp_op_ret;
vcast_cp_mask 扩展含义：
bit0 = 1：ldcstc_accept 存在
bit1 = 1：ldcstc_done 存在
bit2 = 1：ldcstc_read / ldcstc_write 存在

bit3 = 1：write2 存在
bit4 = 1：read2 存在
bit5 = 1：data_process 存在
bit6 = 1：write 存在
bit7 = 1：read 存在
常用值：
vcast_cp_mask = 255U：所有回调都存在
vcast_cp_mask = 0U：所有回调都为空
vcast_cp_op_ret 用于控制：
write2 / read2 / data_process / write / read
这些回调返回 true 还是 false。
 
⸻
 
2. Unit Prefix User Code
用下面这段替换原来的 vcast_get_cp 和 coprocessor 回调，不要重复定义同名函数。
#include <vmm_types.h>
#include <arch_regs.h>
#include <cpu_vcpu_coproc.h>

extern unsigned int vcast_cp_mask;
extern unsigned int vcast_cp_accept_ret;
extern unsigned int vcast_cp_done_after;
extern unsigned int vcast_cp_read_value_seq[8];
extern unsigned int vcast_cp_op_ret;

static struct cpu_vcpu_coproc vcast_cp_obj;

static bool vcast_ldcstc_accept(struct vmm_vcpu *vcpu,
                                arch_regs_t *regs,
                                u32 D,
                                u32 CRd,
                                bool uopt,
                                u32 imm8)
{
    return vcast_cp_accept_ret ? TRUE : FALSE;
}

static bool vcast_ldcstc_done(struct vmm_vcpu *vcpu,
                              arch_regs_t *regs,
                              u32 i,
                              u32 D,
                              u32 CRd,
                              bool uopt,
                              u32 imm8)
{
    return (i >= vcast_cp_done_after) ? TRUE : FALSE;
}

static u32 vcast_ldcstc_read(struct vmm_vcpu *vcpu,
                             arch_regs_t *regs,
                             u32 i,
                             u32 D,
                             u32 CRd,
                             bool uopt,
                             u32 imm8)
{
    return vcast_cp_read_value_seq[i & 7U];
}

static void vcast_ldcstc_write(struct vmm_vcpu *vcpu,
                               arch_regs_t *regs,
                               u32 i,
                               u32 D,
                               u32 CRd,
                               bool uopt,
                               u32 imm8,
                               u32 data)
{
    /* 空实现即可。 */
}

static bool vcast_cp_write2(struct vmm_vcpu *vcpu,
                            arch_regs_t *regs,
                            u32 opc1,
                            u32 CRm,
                            u32 data,
                            u32 data2)
{
    return vcast_cp_op_ret ? TRUE : FALSE;
}

static bool vcast_cp_read2(struct vmm_vcpu *vcpu,
                           arch_regs_t *regs,
                           u32 opc1,
                           u32 CRm,
                           u32 *data,
                           u32 *data2)
{
    if (data) {
        *data = vcast_cp_read_value_seq[0];
    }

    if (data2) {
        *data2 = vcast_cp_read_value_seq[1];
    }

    return vcast_cp_op_ret ? TRUE : FALSE;
}

static bool vcast_cp_data_process(struct vmm_vcpu *vcpu,
                                  arch_regs_t *regs,
                                  u32 opc1,
                                  u32 opc2,
                                  u32 CRd,
                                  u32 CRn,
                                  u32 CRm)
{
    return vcast_cp_op_ret ? TRUE : FALSE;
}

static bool vcast_cp_write(struct vmm_vcpu *vcpu,
                           arch_regs_t *regs,
                           u32 opc1,
                           u32 opc2,
                           u32 CRn,
                           u32 CRm,
                           u32 data)
{
    return vcast_cp_op_ret ? TRUE : FALSE;
}

static bool vcast_cp_read(struct vmm_vcpu *vcpu,
                          arch_regs_t *regs,
                          u32 opc1,
                          u32 opc2,
                          u32 CRn,
                          u32 CRm,
                          u32 *data)
{
    if (data) {
        *data = vcast_cp_read_value_seq[0];
    }

    return vcast_cp_op_ret ? TRUE : FALSE;
}

struct cpu_vcpu_coproc *vcast_get_cp(void)
{
    vcast_cp_obj.ldcstc_accept =
        (vcast_cp_mask & 1U) ? vcast_ldcstc_accept : 0;

    vcast_cp_obj.ldcstc_done =
        (vcast_cp_mask & 2U) ? vcast_ldcstc_done : 0;

    vcast_cp_obj.ldcstc_read =
        (vcast_cp_mask & 4U) ? vcast_ldcstc_read : 0;

    vcast_cp_obj.ldcstc_write =
        (vcast_cp_mask & 4U) ? vcast_ldcstc_write : 0;

    vcast_cp_obj.write2 =
        (vcast_cp_mask & 8U) ? vcast_cp_write2 : 0;

    vcast_cp_obj.read2 =
        (vcast_cp_mask & 16U) ? vcast_cp_read2 : 0;

    vcast_cp_obj.data_process =
        (vcast_cp_mask & 32U) ? vcast_cp_data_process : 0;

    vcast_cp_obj.write =
        (vcast_cp_mask & 64U) ? vcast_cp_write : 0;

    vcast_cp_obj.read =
        (vcast_cp_mask & 128U) ? vcast_cp_read : 0;

    return &vcast_cp_obj;
}
