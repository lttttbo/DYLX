二、
arm_inst_ldcx_l
需要的最小 User Code
arm_inst_ldcx_l 会通过：
cp = cpu_vcpu_coproc_get(coproc);
拿到 struct cpu_vcpu_coproc *，然后访问：
cp->ldcstc_accept
cp->ldcstc_done
cp->ldcstc_write
因此需要一个明确的全局 vcast_cp。
1. Unit Prefix User Code
放在 Unit Prefix User Code，不要放 User Global，因为 struct cpu_vcpu_coproc 在 User Global 位置可能还是不完整类型。
static struct cpu_vcpu_coproc vcast_cp;

static unsigned int vcast_cp_accept_ret;
static unsigned int vcast_cp_done_after;

static unsigned int vcast_cp_accept_call;
static unsigned int vcast_cp_done_call;
static unsigned int vcast_cp_write_call;

static unsigned int vcast_cp_last_D;
static unsigned int vcast_cp_last_CRd;
static unsigned int vcast_cp_last_uopt;
static unsigned int vcast_cp_last_imm8;

static unsigned int vcast_cp_write_i_seq[4];
static unsigned int vcast_cp_write_data_seq[4];

static bool vcast_ldcstc_accept(struct vmm_vcpu *vcpu,
                                arch_regs_t *regs,
                                u32 D,
                                u32 CRd,
                                bool uopt,
                                u32 imm8)
{
    vcast_cp_accept_call++;
    vcast_cp_last_D = D;
    vcast_cp_last_CRd = CRd;
    vcast_cp_last_uopt = uopt ? 1U : 0U;
    vcast_cp_last_imm8 = imm8;
    return vcast_cp_accept_ret ? TRUE : FALSE;
}

static bool vcast_ldcstc_done(struct vmm_vcpu *vcpu,
                              arch_regs_t *regs,
                              u32 i,
                              u32 D,
                              u32 CRd,
                              bool uopt,
                              u32 imm8)
{
    vcast_cp_done_call++;
    return (i >= vcast_cp_done_after) ? TRUE : FALSE;
}

static void vcast_ldcstc_write(struct vmm_vcpu *vcpu,
                               arch_regs_t *regs,
                               u32 i,
                               u32 D,
                               u32 CRd,
                               bool uopt,
                               u32 imm8,
                               u32 data)
{
    if (vcast_cp_write_call < 4U) {
        vcast_cp_write_i_seq[vcast_cp_write_call] = i;
        vcast_cp_write_data_seq[vcast_cp_write_call] = data;
    }
    vcast_cp_write_call++;
}

三、
arm_inst_ldcx_l
字段：
cond   bits[31:28]
P      bit[24]
U      bit[23]
D      bit[22]
W      bit[21]
CRd    bits[15:12]
coproc bits[11:8]
imm8   bits[7:0]
地址计算：
base = arm_align(regs.pc, 4)

imm32 = imm8 << 2

P=1 -> address = offset_addr
P=0 -> address = base

U=1 -> offset_addr = base + imm32
U=0 -> offset_addr = base - imm32
 
⸻
