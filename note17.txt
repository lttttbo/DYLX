下面先约定一个默认构造，后续每条用例的“构造要点”只写差异项：
static struct vmm_vcpu vcpu;
static arch_regs_t regs;
unsigned int inst;

memset(&vcpu, 0, sizeof(vcpu));
memset(&regs, 0, sizeof(regs));

vcpu.is_normal = 1U;
arm_pc(&regs) = 0x1000U;
arm_cpsr(&regs) = 0U;

inst = 0U;
inst |= (0xEU << ARM_INST_COND_SHIFT);
如果某条用例写“条件不成立”，则构造为：
inst = 0U;
inst |= (0x0U << ARM_INST_COND_SHIFT);
arm_cpsr(&regs) = 0U;
