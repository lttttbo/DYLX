static struct vmm_vcpu vcpu;
static arch_regs_t regs;

memset(&vcpu, 0, sizeof(vcpu));

arm_inst_ldr_r.001.R
构造代码
/*
 * cond=AL
 * P=1,U=1,W=0
 * Rn=1,Rt=2,Rm=4
 */
inst = 0xE1812004U;

vcast_reg_read_value[1] = 0x2000U;
vcast_reg_read_value[4] = 0x10U;

vcast_mem_read_value_seq[0] = 0x12345678U;
vcast_mem_read_rc_seq[0] = 0;

<<emulate_arm.arm_inst_ldr_r.inst>> = inst;
<<emulate_arm.arm_inst_ldr_r.regs>> = &regs;
<<emulate_arm.arm_inst_ldr_r.vcpu>> = &vcpu;
期望值
return = 0
regs.pc = 0x1004

vcast_mem_read_call = 1
vcast_mem_read_addr_seq[0] = 0x2010
vcast_mem_read_size_seq[0] = 4

vcast_reg_write_called[1] = 0
vcast_reg_write_called[2] = 1
vcast_reg_write_value[2] = 0x12345678
注释
测试目的：验证 ldr register 在 pre-index add 且不写回时，能够使用 Rm 偏移读取 word 并写入 Rt。 用例类别：基于需求的测试 预置条件：设置 R1 返回 0x2000，R4 返回 0x10；设置内存读取返回 0x12345678。 测试步骤： step1、构造 inst=0xE1812004。 step2、该指令解码后 P=1、U=1、W=0、Rn=1、Rt=2、Rm=4。 step3、调用 arm_inst_ldr_r。 step4、观察返回值、PC、内存读取地址和 Rt 写入值。 用例来源：需求分析、有效等价类 追溯ID：arm_inst_ldr_r 等价列表：register offset、pre-index add、无写回、读取成功。 其它说明：覆盖 index=true、add=true、wback=false 的正常路径。 通过准则：返回 0；PC 为 0x1004；读取地址为 0x2010；R2 写入 0x12345678；R1 不写回。
