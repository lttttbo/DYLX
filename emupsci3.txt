下面是按 64 位 cpu_emulate_psci.h 重新整理后的 emulate_psci.c / psci.c 单元测试构造方案。入参映射统一使用你提醒的单元名：
<<emulate_psci.xxx.yyy>>
不再使用：
<<psci.xxx.yyy>>
 
⸻
 
一、User Code 修改方案
1. Test Case Header User Code
改写目的： 让 Test Case Init / Value User Code 中可以使用 memset，避免隐式声明告警或编译失败。
#include <string.h>
 
⸻
 
2. User Global / Data
改写目的： 仅保存外部 manager stub 所需的简单控制变量。 不在这里定义 struct vmm_vcpu、struct vmm_guest、struct arm_priv 等完整结构体对象，避免不完整类型或作用域问题。
void *vcast_psci_vcpus[4];
unsigned int vcast_psci_vcpu_count;

unsigned int vcast_psci_state[4];

int vcast_psci_vcpu_reset_rc;
int vcast_psci_vcpu_kick_rc;
int vcast_psci_guest_shutdown_rc;
int vcast_psci_guest_reboot_rc;
int vcast_psci_guest_suspend_rc;
 
⸻
 
3. Unit Prefix User Code
改写目的： 统一封装 64 位 PSCI helper 需要访问的字段。当前正确的 helper 使用：
arm_priv(vcpu)->sysregs.mpidr_el1
regs->pstate
arm_priv(vcpu)->sysregs.sctlr_el1
所以这里统一定义宏，后续用例构造更清晰。
#define VCAST_SET_MPIDR(_priv, _val) \
    ((_priv).sysregs.mpidr_el1 = (u64)(_val))

#define VCAST_SET_MODE32(_regs) \
    ((_regs).pstate |= (u64)PSR_MODE32)

#define VCAST_SET_BE32(_regs) \
    ((_regs).pstate |= ((u64)PSR_MODE32 | (u64)CPSR_BE_ENABLED))

#define VCAST_GET_THUMB(_regs) \
    ((_regs).pstate & (u64)CPSR_THUMB_ENABLED)

#define VCAST_GET_BE32(_regs) \
    ((_regs).pstate & (u64)CPSR_BE_ENABLED)

#define VCAST_GET_BE64(_priv) \
    ((_priv).sysregs.sctlr_el1 & (u64)SCTLR_EE_MASK)
说明：
VCAST_SET_MPIDR：用于真实执行 emulate_psci_get_mpidr。
VCAST_SET_MODE32：用于真实执行 emulate_psci_is_32bit。
VCAST_SET_BE32：用于真实执行 emulate_psci_is_be 的 32-bit 分支，并避免执行 mrs(sctlr_el1)。
VCAST_GET_THUMB：用于检查 target VCPU 是否被设置 Thumb。
VCAST_GET_BE32：用于检查 target 32-bit VCPU 是否被设置 BE。
VCAST_GET_BE64：用于检查 target 64-bit VCPU 是否通过 sctlr_el1 设置 BE。
 
⸻
 
4. Stub：
vmm_manager_guest_next_vcpu
改写目的： 保留源码真实宏：
vmm_manager_for_each_guest_vcpu
不再重定义这个宏。只 stub 其调用的外部 manager 遍历函数，使遍历顺序可控，避免真实链表、锁和全局 manager 状态。
unsigned int i;

if (!guest) {
    return (struct vmm_vcpu *)0;
}

if (!current_vcpu) {
    if (vcast_psci_vcpu_count == 0U) {
        return (struct vmm_vcpu *)0;
    }

    return (struct vmm_vcpu *)vcast_psci_vcpus[0];
}

for (i = 0U; i < vcast_psci_vcpu_count; i++) {
    if (current_vcpu == (struct vmm_vcpu *)vcast_psci_vcpus[i]) {
        if ((i + 1U) < vcast_psci_vcpu_count) {
            return (struct vmm_vcpu *)vcast_psci_vcpus[i + 1U];
        }

        return (struct vmm_vcpu *)0;
    }
}

return (struct vmm_vcpu *)0;
 
⸻
 
5. Stub：
vmm_manager_vcpu_get_state
改写目的： 隔离真实 manager 状态，通过数组为不同 VCPU 提供可控状态。
unsigned int i;

for (i = 0U; i < vcast_psci_vcpu_count; i++) {
    if (vcpu == (struct vmm_vcpu *)vcast_psci_vcpus[i]) {
        return vcast_psci_state[i];
    }
}

return VMM_VCPU_STATE_UNKNOWN;
 
⸻
 
6. 其它外部副作用函数 stub
这些函数属于 manager、副作用、等待中断、打印边界，保留 stub 是合理的。
vmm_manager_vcpu_reset
改写目的： 隔离真实 reset 副作用，通过返回值控制成功 / 失败分支。
return vcast_psci_vcpu_reset_rc;
vmm_manager_vcpu_kick
改写目的： 隔离真实调度唤醒副作用，通过返回值控制成功 / 失败分支。
return vcast_psci_vcpu_kick_rc;
vmm_manager_guest_shutdown_request
改写目的： 隔离真实 guest shutdown 副作用。
return vcast_psci_guest_shutdown_rc;
vmm_manager_guest_reboot_request
改写目的： 隔离真实 guest reboot 副作用。
return vcast_psci_guest_reboot_rc;
vmm_manager_guest_suspend_request
改写目的： 隔离真实 guest suspend 副作用。
return vcast_psci_guest_suspend_rc;
vmm_vcpu_irq_wait_timeout
改写目的： 避免单元测试中真实等待中断。
return;
vmm_printf
改写目的： 避免打印依赖和格式化副作用，使错误分支可运行。
return 0;
 
⸻

 * 高 32 bit 非 0，用于覆盖 32-bit cpu_id mask。

regs.gpr[3] = 0x55AAUL;

vcast_psci_vcpu_kick_rc = 0;

<<emulate_psci.psci_vcpu_on.source_vcpu>> = &src;
<<emulate_psci.psci_vcpu_on.regs>> = &regs;

return = 0
tgt.regs.pc = 0x8000
tgt.regs.gpr[0] = 0x55AA
可选期望：
VCAST_GET_THUMB(tgt.regs) != 0
VCAST_GET_BE32(tgt.regs) != 0
注释
测试目的：验证 CPU_ON 在找到目标 VCPU、目标处于 RESET、source 为 32-bit BE、目标为 32-bit、Thumb entry 且 kick 成功时返回成功。 用例类别：基于需求的测试 预置条件：构造 source 和 target 两个 VCPU；target mpidr 与 cpu_id 低 32 位匹配；target 状态为 RESET。 测试步骤： step1、配置 source 和 target 的 guest、arch private、mpidr。 step2、配置 VCPU 遍历数组包含 source 和 target。 step3、设置 target 状态为 RESET。 step4、设置 source regs 为 PSR_MODE32 | CPSR_BE_ENABLED。 step5、设置 target regs 为 PSR_MODE32。 step6、设置 regs.gpr[1]=0xFFFF000000000001。 step7、设置 regs.gpr[2]=0x8001。 step8、设置 regs.gpr[3]=0x55AA。 step9、设置 vmm_manager_vcpu_kick 返回成功。 step10、调用 psci_vcpu_on。 step11、观察返回值、target PC 和 target r0。 用例来源：需求分析 追溯ID：psci_vcpu_on 等价列表：CPU_ON 成功路径、32-bit mask、Thumb entry、BE32 继承。 其它说明：覆盖 emulate_psci_is_32bit 为 true、cpu_id mask、target 匹配、target RESET、Thumb 设置、BE32 设置、kick 成功。 通过准则：函数返回 0；target PC 为 0x8000；target r0 为 0x55AA。

十六、负值期望说明
本文件中有两种返回方式：
1. 函数直接返回 
unsigned long
例如：
psci_vcpu_off
psci_vcpu_on
psci_vcpu_affinity_info
do_psci_1_0_features
psci_system_suspend
负 PSCI 返回值按 64-bit unsigned 显示：
-1 -> 0xFFFFFFFFFFFFFFFF
-2 -> 0xFFFFFFFFFFFFFFFE
-3 -> 0xFFFFFFFFFFFFFFFD
-4 -> 0xFFFFFFFFFFFFFFFC
-6 -> 0xFFFFFFFFFFFFFFFA
2. 
emulate_psci_0_1_call
/ 
emulate_psci_0_2_call
写回 
regs.gpr[0]
64 位 helper 中：
emulate_psci_set_reg(..., (u64)val)
所以负 PSCI 返回值写到 regs.gpr[0] 后同样是 64-bit 形式：
PSCI_RET_NOT_SUPPORTED    -> 0xFFFFFFFFFFFFFFFF
PSCI_RET_DENIED           -> 0xFFFFFFFFFFFFFFFD
PSCI_RET_INTERNAL_FAILURE -> 0xFFFFFFFFFFFFFFFA
emulate_psci_call 自身返回 int，所以：
VMM_OK       -> 0
VMM_EINVALID -> -5
不变。
