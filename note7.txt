下面按当前模板整理 vmm_workqueue.c：
	•	每个函数独立设计；
	•	给出具体构造方法；
	•	给出 Expected return 和关键期望；
	•	注释中步骤分行写；
	•	workqueue_main 是无限循环函数，单独给受控退出方案；
	•	部分同文件调用建议用 SBF，否则会变成其它函数的间接覆盖。

⸻

一、公共 fixture 设置

struct vmm_workqueue 和 wqctrl 都是 vmm_workqueue.c 内部定义，Value User Code 侧通常不能直接定义或展开。因此建议把对象定义和少量构造函数放到 Unit Appendix User Code -> vmm_workqueue。

1. Unit Appendix User Code -> vmm_workqueue

/*
 * VectorCAST workqueue fixture.
 * struct vmm_workqueue and wqctrl are private to vmm_workqueue.c,
 * so object definitions and list manipulation helpers are placed here.
 */

static struct vmm_workqueue vcast_wq1_obj;
static struct vmm_workqueue vcast_wq2_obj;
static struct vmm_thread vcast_wq_thread1_obj;
static struct vmm_thread vcast_wq_thread2_obj;

/* Direct pointer handles used by Value User Code */
struct vmm_workqueue *vcast_wq1 = &vcast_wq1_obj;
struct vmm_workqueue *vcast_wq2 = &vcast_wq2_obj;
struct vmm_thread *vcast_wq_thread1 = &vcast_wq_thread1_obj;
struct vmm_thread *vcast_wq_thread2 = &vcast_wq_thread2_obj;

static void vcast_wq_zero(void *ptr, unsigned long size)
{
    unsigned char *p = (unsigned char *)ptr;

    while (size > 0UL) {
        *p = 0U;
        ++p;
        --size;
    }
}

void vcast_wq_fixture_reset(void)
{
    vcast_wq_zero(&vcast_wq1_obj, sizeof(vcast_wq1_obj));
    vcast_wq_zero(&vcast_wq2_obj, sizeof(vcast_wq2_obj));
    vcast_wq_zero(&vcast_wq_thread1_obj, sizeof(vcast_wq_thread1_obj));
    vcast_wq_zero(&vcast_wq_thread2_obj, sizeof(vcast_wq_thread2_obj));
    vcast_wq_zero(&wqctrl, sizeof(wqctrl));

    INIT_SPIN_LOCK(&vcast_wq1_obj.lock);
    INIT_LIST_HEAD(&vcast_wq1_obj.head);
    INIT_LIST_HEAD(&vcast_wq1_obj.work_list);
    INIT_COMPLETION(&vcast_wq1_obj.work_avail);
    vcast_wq1_obj.thread = &vcast_wq_thread1_obj;

    INIT_SPIN_LOCK(&vcast_wq2_obj.lock);
    INIT_LIST_HEAD(&vcast_wq2_obj.head);
    INIT_LIST_HEAD(&vcast_wq2_obj.work_list);
    INIT_COMPLETION(&vcast_wq2_obj.work_avail);
    vcast_wq2_obj.thread = &vcast_wq_thread2_obj;

    INIT_SPIN_LOCK(&wqctrl.lock);
    INIT_LIST_HEAD(&wqctrl.wq_list);
    wqctrl.wq_count = 0U;
}

void vcast_wq_global_empty(void)
{
    vcast_wq_fixture_reset();
}

void vcast_wq_global_one(void)
{
    vcast_wq_fixture_reset();

    list_add_tail(&vcast_wq1_obj.head, &wqctrl.wq_list);
    wqctrl.wq_count = 1U;
}

void vcast_wq_global_two(void)
{
    vcast_wq_fixture_reset();

    list_add_tail(&vcast_wq1_obj.head, &wqctrl.wq_list);
    list_add_tail(&vcast_wq2_obj.head, &wqctrl.wq_list);
    wqctrl.wq_count = 2U;
}

void vcast_wq_set_count(unsigned int count)
{
    wqctrl.wq_count = count;
}

void vcast_wq_install_syswq0(void)
{
    wqctrl.syswq[0] = &vcast_wq1_obj;
}

void vcast_wq_add_work_to_wq1(struct vmm_work *work)
{
    if (!work) {
        return;
    }

    INIT_LIST_HEAD(&work->head);
    work->wq = &vcast_wq1_obj;

    list_add_tail(&work->head, &vcast_wq1_obj.work_list);
}

void vcast_wq_clear_work_list(struct vmm_workqueue *wq)
{
    if (wq) {
        INIT_LIST_HEAD(&wq->work_list);
    }
}
