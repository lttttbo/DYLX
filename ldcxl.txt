可以复用，但 stcx 精简版 usercode 不能原封不动用于 ldcx_l，因为：
arm_inst_stcx 需要 cp->ldcstc_read
arm_inst_ldcx_l 需要 cp->ldcstc_write
所以最小修改是：保留你现在的 vcast_cp_mask / vcast_cp_accept_ret / vcast_cp_done_after 控制变量，再补一个 空实现的 ldcstc_write 回调，并让 vcast_cp_mask 的 bit2 同时控制：
stcx 用的 ldcstc_read
ldcx_l 用的 ldcstc_write
这样同一套 usercode 可以同时覆盖 stcx 和 ldcx_l。
 
⸻
 
1. 修改后的精简 User Code
1.1 User Global / Data
如果你已经有这些变量，不要重复定义。缺哪个补哪个。
unsigned int vcast_cp_mask;
unsigned int vcast_cp_accept_ret;
unsigned int vcast_cp_done_after;
unsigned int vcast_cp_read_value_seq[8];
vcast_cp_mask 含义保持不变：
bit0 = 1：ldcstc_accept 存在
bit1 = 1：ldcstc_done 存在
bit2 = 1：ldcstc_read / ldcstc_write 存在

vcast_cp_mask = 7：accept、done、read/write 全部存在
vcast_cp_mask = 6：accept 为 NULL
vcast_cp_mask = 5：done 为 NULL
vcast_cp_mask = 3：read/write 为 NULL
 
⸻
 
1.2 Unit Prefix User Code
如果你已经放过 stcx 的精简版，只需要确认里面有下面这个 vcast_ldcstc_write，以及 vcast_get_cp() 里设置了 ldcstc_write。
#include <vmm_types.h>
#include <arch_regs.h>
#include <cpu_vcpu_coproc.h>

extern unsigned int vcast_cp_mask;
extern unsigned int vcast_cp_accept_ret;
extern unsigned int vcast_cp_done_after;
extern unsigned int vcast_cp_read_value_seq[8];

static struct cpu_vcpu_coproc vcast_cp_obj;

static bool vcast_ldcstc_accept(struct vmm_vcpu *vcpu,
                                arch_regs_t *regs,
                                u32 D,
                                u32 CRd,
                                bool uopt,
                                u32 imm8)
{
    return vcast_cp_accept_ret ? TRUE : FALSE;
}

static bool vcast_ldcstc_done(struct vmm_vcpu *vcpu,
                              arch_regs_t *regs,
                              u32 i,
                              u32 D,
                              u32 CRd,
                              bool uopt,
                              u32 imm8)
{
    return (i >= vcast_cp_done_after) ? TRUE : FALSE;
}

static u32 vcast_ldcstc_read(struct vmm_vcpu *vcpu,
                             arch_regs_t *regs,
                             u32 i,
                             u32 D,
                             u32 CRd,
                             bool uopt,
                             u32 imm8)
{
    return vcast_cp_read_value_seq[i & 7U];
}

static void vcast_ldcstc_write(struct vmm_vcpu *vcpu,
                               arch_regs_t *regs,
                               u32 i,
                               u32 D,
                               u32 CRd,
                               bool uopt,
                               u32 imm8,
                               u32 data)
{
    /* 空实现即可。ldcx_l 只需要该回调存在来覆盖路径。 */
}

struct cpu_vcpu_coproc *vcast_get_cp(void)
{
    vcast_cp_obj.ldcstc_accept =
        (vcast_cp_mask & 1U) ? vcast_ldcstc_accept : 0;

    vcast_cp_obj.ldcstc_done =
        (vcast_cp_mask & 2U) ? vcast_ldcstc_done : 0;

    vcast_cp_obj.ldcstc_read =
        (vcast_cp_mask & 4U) ? vcast_ldcstc_read : 0;

    vcast_cp_obj.ldcstc_write =
        (vcast_cp_mask & 4U) ? vcast_ldcstc_write : 0;

    return &vcast_cp_obj;
}
