/*
 * 改写目的：
 * 模拟 vmm_manager_init() 为 mngr.guest_array 和
 * mngr.guest_avail_array 建立有效存储。
 *
 * 默认将全部 guest 槽位设置为可用状态。
 * 每条用例再将需要使用的槽位置为 FALSE。
 */
void vcast_manager_prepare_guest_env(u32 guest_count)
{
    u32 g;

    memset(vcast_guest_array,
           0,
           sizeof(vcast_guest_array));

    for (g = 0U;
         g < (u32)CONFIG_MAX_GUEST_COUNT;
         ++g) {
        vcast_guest_array[g].id = g;
        vcast_guest_avail_array[g] = TRUE;
    }

    mngr.guest_array = vcast_guest_array;
    mngr.guest_avail_array = vcast_guest_avail_array;
    mngr.guest_count = guest_count;
}
该函数本身会被统计，因此后面提供一条独立用例。
