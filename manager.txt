static int vcast_vcpu_iter_cb(struct vmm_vcpu *vcpu, void *priv)
{
    vcast_vcpu_iter_count++;
    vcast_vcpu_iter_last_priv = priv;

    if (vcpu) {
        vcast_vcpu_iter_last_id = vcpu->id;
    } else {
        vcast_vcpu_iter_last_id = 0xFFFFFFFFU;
    }

    if ((vcast_vcpu_iter_break_after != 0U) &&
        (vcast_vcpu_iter_count >= vcast_vcpu_iter_break_after)) {
        return vcast_vcpu_iter_rc;
    }

    return VMM_OK;
}
