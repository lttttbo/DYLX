static int number(const char *s, u64 *n)
{
	*n = 0;
	if (!*s) return VMM_EINVALID;
	while (*s) {
		if (*s < '0' || *s > '9' || *n > (~0ULL - (*s - '0')) / 10)
			return VMM_EINVALID;
		*n = *n * 10 + (*s++ - '0');
	}
	return VMM_OK;
}

{
	vmm_cprintf(cdev, "\"%s\":{\"count\":%llu,\"mean_ns\":%llu,"
		"\"p99_upper_ns\":%llu,\"max_ns\":%llu,\"jitter_range_ns\":%llu}",
		name, MON_ULL(s->count), MON_ULL(s->mean_ns), MON_ULL(s->p99_upper_ns),
		MON_ULL(s->max_ns), MON_ULL(s->max_ns - s->min_ns));
}

static void guest_status(struct vmm_chardev *cdev, u32 id)
{
	struct vmm_monitor_guest r;
	u64 now = vmm_timer_timestamp();
	if (vmm_monitor_guest_get(id, &r)) {
		vmm_cprintf(cdev, "{\"schema\":2,\"kind\":\"guest_os\",\"guest\":%u,\"quality\":\"missing\"}\n", id);
		return;
	}
	vmm_cprintf(cdev, "{\"schema\":2,\"kind\":\"guest_os\",\"guest\":%u,\"session\":%llu,\"seq\":%llu,"
		"\"received_ns\":%llu,\"age_ns\":%llu,\"quality\":\"%s\",\"valid\":%u,\"cpu_bp\":%u,"
		"\"total_bytes\":%llu,\"available_bytes\":%llu,\"free_bytes\":%llu,\"used_estimate_bytes\":%llu}\n",
		id, MON_ULL(r.session), MON_ULL(r.sequence), MON_ULL(r.received_ns),
		MON_ULL(now >= r.received_ns ? now - r.received_ns : 0), !vmm_monitor_enabled() ? "disabled" :
		now < r.received_ns || now - r.received_ns > 3 * VMM_MONITOR_NS ? "stale" : "valid",
		r.valid, r.cpu_bp, MON_ULL(r.total_bytes), MON_ULL(r.available_bytes), MON_ULL(r.free_bytes),
		MON_ULL(r.total_bytes - r.available_bytes));
}

static int diagnose(struct vmm_chardev *cdev, u32 ms)
{
	u32 i, ec, suspect;
	int rc;
	struct vmm_monitor_diag_record r;
	vmm_mutex_lock(&diagnose_lock);
	rc = vmm_monitor_diag_start(ms);
	if (rc) goto done;
	vmm_msleep(ms);
	vmm_monitor_diag_stop();
	for (i = 0; i < VMM_MONITOR_DIAG_SLOTS; i++) {
		if (vmm_monitor_diag_get(i, &r)) continue;
		ec = (r.esr >> 26) & 63;
		/* WFI/WFE is excluded from repetition/latency suspicion. */
		suspect = r.errors ? 4 : 0;
		if (ec != 1) {
			if (r.count * 1000 > (u64)1000 * ms) suspect |= 1;
			if (r.max_ns > 1000000) suspect |= 2;
		}
		vmm_cprintf(cdev, "{\"schema\":2,\"kind\":\"trap_diagnostic\",\"guest\":%u,\"vcpu\":%u,"
			"\"elr\":%llu,\"esr\":%llu,\"ec\":%u,\"count\":%llu,\"mean_service_ns\":%llu,"
			"\"max_service_ns\":%llu,\"last_ns\":%llu,\"errors\":%u,\"suspect\":%u}\n",
			r.guest, r.vcpu, MON_ULL(r.pc), MON_ULL(r.esr), ec, MON_ULL(r.count),
			MON_ULL(udiv64(r.sum_ns, r.count)), MON_ULL(r.max_ns), MON_ULL(r.last_ns), r.errors, suspect);
	}
	vmm_cprintf(cdev, "{\"schema\":2,\"kind\":\"diagnostic_summary\",\"window_ms\":%u,"
		"\"untracked\":%llu,\"pmu_cycles\":\"not_collected\"}\n", ms, MON_ULL(vmm_monitor_diag_overflow()));
	/* Preserve the product's optional diagnostic extension after built-in data. */
	rc = vmm_monitor_platform_diagnose(cdev);
	if (rc == VMM_ENOTAVAIL) rc = VMM_OK;
done:
	vmm_mutex_unlock(&diagnose_lock);
	return rc;
}

static int monitor_exec(struct vmm_chardev *cdev, int argc, char **argv)
{
	u32 i;
	u64 value, seq, now;
	int rc;
	struct vmm_monitor_status st;
	struct vmm_monitor_snapshot s;
	struct vmm_monitor_event e;
	if (argc < 2) goto usage;
	if (argc == 2 && !strcmp(argv[1], "start")) return vmm_monitor_enable(TRUE);
	if (argc == 2 && !strcmp(argv[1], "stop")) return vmm_monitor_enable(FALSE);
	if ((argc == 2 || argc == 3) && !strcmp(argv[1], "diagnose")) {
		value = 1000;
		if (argc == 3 && number(argv[2], &value)) return VMM_EINVALID;
		return value && value <= 5000 ? diagnose(cdev, value) : VMM_EINVALID;
	}
	vmm_monitor_status(&st);
	if (argc == 4 && !strcmp(argv[1], "config")) {
		if (number(argv[3], &value)) return VMM_EINVALID;
		if (!strcmp(argv[2], "ready_ns")) st.config.ready_limit_ns = value;
		else if (!strcmp(argv[2], "irq_ns")) st.config.irq_limit_ns = value;
		else if (!strcmp(argv[2], "exit_rate") && value <= 1000000000) st.config.exit_rate = value;
		else if (!strcmp(argv[2], "sustain_seconds") && value <= 60) st.config.sustain_seconds = value;
		else if (!strcmp(argv[2], "log") && value <= 1) st.config.log = value;
		else if (!strcmp(argv[2], "publish") && value <= 1) st.config.publish = value;
		else if (!strcmp(argv[2], "cpu_bp") && value <= 10000) st.config.guest_cpu_limit_bp = value;
		else return VMM_EINVALID;
		return vmm_monitor_configure(&st.config);
	}
	if (argc == 3 && !strcmp(argv[1], "mem")) {
		struct vmm_guest *guest = vmm_manager_guest_find(argv[2]);
		char *args[] = { "guest", "region_list", argv[2] };
		if (!guest) return VMM_ENOTAVAIL;
		guest_status(cdev, guest->id);
		return vmm_cmdmgr_execute_cmd(cdev, 3, args);
	}
	if (argc != 2) goto usage;
	if (!strcmp(argv[1], "guest") || !strcmp(argv[1], "cpu")) {
		for (i = 0; i < CONFIG_MAX_GUEST_COUNT; i++)
			if (vmm_manager_guest(i)) guest_status(cdev, i);
		if (!strcmp(argv[1], "guest")) return VMM_OK;
	}
	if (!strcmp(argv[1], "status")) {
		vmm_cprintf(cdev, "{\"schema\":2,\"enabled\":%u,\"samples\":%llu,"
			"\"late_samples\":%llu,\"sample_max_ns\":%llu,\"events\":%llu,"
			"\"overwritten\":%llu,\"publish_errors\":%llu,\"work_errors\":%llu,"
			"\"irq_untracked\":%llu,\"exit_rate\":%u,\"sustain_seconds\":%u,"
			"\"ready_ns\":%llu,\"irq_ns\":%llu,\"log\":%u,\"publish\":%u,\"busy_snapshots\":%llu,\"cpu_bp\":%u}\n",
			st.enabled, MON_ULL(st.samples), MON_ULL(st.late_samples), MON_ULL(st.sample_max_ns),
			MON_ULL(st.events), MON_ULL(st.overwritten), MON_ULL(st.publish_errors), MON_ULL(st.work_errors),
			MON_ULL(st.irq_untracked), st.config.exit_rate, st.config.sustain_seconds,
			MON_ULL(st.config.ready_limit_ns), MON_ULL(st.config.irq_limit_ns), st.config.log, st.config.publish,
			MON_ULL(st.busy_snapshots), st.config.guest_cpu_limit_bp);
		return VMM_OK;
	}
	if (!strcmp(argv[1], "events")) {
		seq = st.events > VMM_MONITOR_EVENTS ? st.events - VMM_MONITOR_EVENTS + 1 : 1;
		for (; seq <= st.events; seq++) {
			if (vmm_monitor_event_get(seq, &e)) continue;
			vmm_cprintf(cdev, "{\"schema\":2,\"seq\":%llu,\"time_ns\":%llu,\"type\":%u,"
				"\"guest\":%u,\"vcpu\":%u,\"generation\":%u,\"value\":%llu,\"detail\":%llu,\"code\":%u}\n",
				MON_ULL(e.sequence), MON_ULL(e.timestamp_ns), e.type, e.guest_id,
				e.vcpu_id, e.generation, MON_ULL(e.value), MON_ULL(e.detail), e.code);
		}
		return VMM_OK;
	}
	if (!strcmp(argv[1], "cpu") || !strcmp(argv[1], "irq") || !strcmp(argv[1], "collect")) {
		now = vmm_timer_timestamp();
		for (i = 0; i <= CONFIG_MAX_VCPU_COUNT; i++) {
			if (vmm_monitor_snapshot(i, &s)) continue;
			vmm_cprintf(cdev, "{\"schema\":2,\"guest\":%u,\"vcpu\":%u,\"generation\":%u,"
				"\"hcpu\":%u,\"state\":%u,\"quality\":\"%s\",\"time_ns\":%llu,\"interval_ns\":%llu,"
				"\"ready_age_ns\":%llu,\"exits\":%llu,\"exit_rate\":%llu,\"last_exit\":%u,",
				s.guest_id, s.vcpu_id, s.generation, s.hcpu, s.state,
				!st.enabled ? "disabled" : !s.timestamp_ns ? "warmup" :
				now - s.timestamp_ns > 2 * VMM_MONITOR_NS ? "stale" : s.valid ? "valid" : "warmup_or_gap",
				MON_ULL(s.timestamp_ns), MON_ULL(s.interval_ns), MON_ULL(s.ready_age_ns),
				MON_ULL(s.exits), MON_ULL(s.exit_rate), s.last_exit);
			latency(cdev, "ready", &s.ready);
			vmm_cprintf(cdev, ",");
			latency(cdev, "host_irq_to_lr", &s.irq);
			vmm_cprintf(cdev, ",");
			latency(cdev, "dispatch", &s.dispatch);
			vmm_cprintf(cdev, ",\"faults_pending_summary\":%llu,\"last_address\":%llu,"
				"\"last_fault\":%u,\"last_detail\":%llu}\n", MON_ULL(s.faults),
				MON_ULL(s.last_address), s.last_fault, MON_ULL(s.last_detail));
		}
		if (!strcmp(argv[1], "collect")) {
			rc = diagnose(cdev, 1000);
			if (rc) return rc;
		}
		return VMM_OK;
	}
usage:
	monitor_usage(cdev);
	return argc == 2 && !strcmp(argv[1], "help") ? VMM_OK : VMM_EINVALID;
}

static struct vmm_cmd monitor_cmd = {
	.name = "monitor", .desc = "Lightweight hypervisor monitoring",
	.usage = monitor_usage, .exec = monitor_exec,
};
static int __init monitor_cmd_init(void) { return vmm_cmdmgr_register_cmd(&monitor_cmd); }
static void __exit monitor_cmd_exit(void) { vmm_cmdmgr_unregister_cmd(&monitor_cmd); }
VMM_DECLARE_MODULE("Monitor command", "Xvisor contributors", "GPL", 0,
		   monitor_cmd_init, monitor_cmd_exit);

/* No retained vCPU pointers, no allocation in any collection/report path. */
struct monitor_data {
	struct vmm_monitor_snapshot snap;
	struct vmm_monitor_hist ready, irq, dispatch;
	struct vmm_monitor_streak storm;
	u64 epoch_ns, ready_since, previous_ns, exits, faults;
	u64 notice_ns[3];
	bool waiting;
};
struct monitor_slot {
	vmm_spinlock_t lock;
	struct monitor_data data;
};
/* Extra slot records faults with no attributable Guest (e.g. host shmem API). */
static struct monitor_slot slots[CONFIG_MAX_VCPU_COUNT + 1];
static DEFINE_SPINLOCK(control_lock);
static DEFINE_SPINLOCK(sample_lock);
static struct vmm_mutex drain_lock;
static atomic_t initialized = { 0 }, enabled = { 0 };
static struct vmm_monitor_status status;
static struct vmm_monitor_event events[VMM_MONITOR_EVENTS];
static u64 report_sequence = 1;
extern void vmm_monitor_irq_init(void);
extern void vmm_monitor_irq_restart(u64 now);
#ifndef CONFIG_MONITOR_EXTERNAL_EXECUTOR
static struct vmm_workqueue *report_queue;
static struct vmm_work report_work;
static struct vmm_timer_event sample_timer;
static u64 next_sample;
static atomic_t executor_state = {0}, timer_start_result = {VMM_EINPROGRESS};
#endif

bool vmm_monitor_enabled(void)
{
	return arch_atomic_read(&enabled) != 0;
}

static struct monitor_slot *slot_for(struct vmm_vcpu *vcpu)
{
	if (!arch_atomic_read(&initialized) || !vcpu || !vcpu->is_normal ||
	    vcpu->id >= CONFIG_MAX_VCPU_COUNT)
		return NULL;
	return &slots[vcpu->id];
}

static void identify(struct monitor_data *d, struct vmm_vcpu *vcpu)
{
	d->snap.present = TRUE;
	d->snap.vcpu_id = vcpu ? vcpu->id : VMM_MONITOR_UNKNOWN;
	d->snap.guest_id = vcpu ? vcpu->guest->id : VMM_MONITOR_UNKNOWN;
}

static void reset_data(struct monitor_data *d, u64 now)
{
	u32 generation = d->snap.generation + 1;
	memset(d, 0, sizeof(*d));
	d->snap.generation = generation;
	d->epoch_ns = now;
}

void vmm_monitor_state(struct vmm_vcpu *vcpu, u32 state, u64 now)
{
	irq_flags_t flags;
	struct monitor_slot *s = slot_for(vcpu);
	struct monitor_data *d;
	if (!s)
		return;
	/* Lifecycle must invalidate cached data even while collection is off. */
	if (!vmm_monitor_enabled() && state != VMM_VCPU_STATE_RESET &&
	    state != VMM_VCPU_STATE_UNKNOWN)
		return;
	vmm_spin_lock_irqsave_lite(&s->lock, flags);
	d = &s->data;
	if (state == VMM_VCPU_STATE_RESET || state == VMM_VCPU_STATE_UNKNOWN)
		reset_data(d, now);
	identify(d, vcpu);
	d->snap.present = state != VMM_VCPU_STATE_UNKNOWN;
	d->snap.state = state;
	d->snap.hcpu = vcpu->hcpu;
	if (vmm_monitor_enabled()) {
		if (state == VMM_VCPU_STATE_READY && !d->waiting) {
			d->ready_since = now;
			d->waiting = TRUE;
		} else if (state != VMM_VCPU_STATE_READY) {
			if (state == VMM_VCPU_STATE_RUNNING && d->waiting &&
			    now >= d->ready_since)
				vmm_monitor_hist_add(&d->ready, now - d->ready_since);
			d->waiting = FALSE;
		}
	}
	vmm_spin_unlock_irqrestore_lite(&s->lock, flags);
	if (state == VMM_VCPU_STATE_RESET || state == VMM_VCPU_STATE_UNKNOWN)
		vmm_monitor_irq_reset(vcpu);
	if (state == VMM_VCPU_STATE_RESET || state == VMM_VCPU_STATE_UNKNOWN)
		vmm_monitor_guest_reset(vcpu->guest->id);
}

void vmm_monitor_selected(struct vmm_vcpu *vcpu, u64 now)
{
	irq_flags_t flags;
	struct monitor_slot *s;
	if (!vmm_monitor_enabled() || !(s = slot_for(vcpu))) return;
	vmm_spin_lock_irqsave_lite(&s->lock, flags);
	if (vmm_monitor_enabled() && s->data.waiting) {
		if (now >= s->data.ready_since)
			vmm_monitor_hist_add(&s->data.ready, now - s->data.ready_since);
		s->data.waiting = FALSE;
	}
	vmm_spin_unlock_irqrestore_lite(&s->lock, flags);
}

void vmm_monitor_dispatch(struct vmm_vcpu *vcpu, u64 begin, u64 end)
{
	irq_flags_t flags;
	struct monitor_slot *s;
	if (!vmm_monitor_enabled() || end < begin || !(s = slot_for(vcpu))) return;
	vmm_spin_lock_irqsave_lite(&s->lock, flags);
	if (vmm_monitor_enabled()) vmm_monitor_hist_add(&s->data.dispatch, end - begin);
	vmm_spin_unlock_irqrestore_lite(&s->lock, flags);
}

void vmm_monitor_exit(struct vmm_vcpu *vcpu, u32 reason)
{
	irq_flags_t flags;
	struct monitor_slot *s;
	if (!vmm_monitor_enabled() || !(s = slot_for(vcpu)))
		return;
	vmm_spin_lock_irqsave_lite(&s->lock, flags);
	if (vmm_monitor_enabled()) {
		identify(&s->data, vcpu);
		s->data.exits++;
		s->data.snap.last_exit = reason;
	}
	vmm_spin_unlock_irqrestore_lite(&s->lock, flags);
}

void vmm_monitor_mem_fault(struct vmm_vcpu *actor, u32 kind,
			   u64 address, u64 detail)
{
	irq_flags_t flags;
	struct monitor_slot *s;
	if (!vmm_monitor_enabled())
		return;
	s = actor ? slot_for(actor) : &slots[CONFIG_MAX_VCPU_COUNT];
	if (!s)
		return;
	vmm_spin_lock_irqsave_lite(&s->lock, flags);
	if (vmm_monitor_enabled()) {
		identify(&s->data, actor);
		s->data.faults++;
		s->data.snap.last_address = address;
		s->data.snap.last_detail = detail;
		s->data.snap.last_fault = kind;
	}
	vmm_spin_unlock_irqrestore_lite(&s->lock, flags);
}

void vmm_monitor_irq_sample(struct vmm_vcpu *vcpu, u64 start_ns, u64 end_ns)
{
	irq_flags_t flags;
	struct monitor_slot *s;
	if (!vmm_monitor_enabled() || !(s = slot_for(vcpu)))
		return;
	vmm_spin_lock_irqsave_lite(&s->lock, flags);
	if (vmm_monitor_enabled() && start_ns >= s->data.epoch_ns &&
	    end_ns >= start_ns) {
		identify(&s->data, vcpu);
		vmm_monitor_hist_add(&s->data.irq, end_ns - start_ns);
	}
	vmm_spin_unlock_irqrestore_lite(&s->lock, flags);
}

static void record_event(struct vmm_monitor_snapshot *s, u32 type,
			 u64 value, u64 detail, u32 code)
{
	irq_flags_t flags;
	struct vmm_monitor_event *e;
	vmm_spin_lock_irqsave_lite(&control_lock, flags);
	status.events++;
	if (status.events - report_sequence >= VMM_MONITOR_EVENTS) {
		report_sequence++;
		status.overwritten++;
	}
	e = &events[(status.events - 1) % VMM_MONITOR_EVENTS];
	e->sequence = status.events;
	e->timestamp_ns = s->timestamp_ns;
	e->guest_id = s->guest_id;
	e->vcpu_id = s->vcpu_id;
	e->generation = s->generation;
	e->type = type;
	e->value = value;
	e->detail = detail;
	e->code = code;
	vmm_spin_unlock_irqrestore_lite(&control_lock, flags);
}

void vmm_monitor_guest_event(u32 guest, u64 now, u64 value, u32 limit)
{
	struct vmm_monitor_snapshot s = {0};
	s.guest_id = guest; s.vcpu_id = VMM_MONITOR_UNKNOWN; s.timestamp_ns = now;
	record_event(&s, VMM_MONITOR_GUEST_CPU_HIGH, value, limit, 0);
}

void vmm_monitor_status(struct vmm_monitor_status *out)
{
	irq_flags_t flags;
	vmm_spin_lock_irqsave_lite(&control_lock, flags);
	*out = status;
	out->enabled = vmm_monitor_enabled();
	out->irq_untracked = vmm_monitor_irq_untracked();
	vmm_spin_unlock_irqrestore_lite(&control_lock, flags);
}

int vmm_monitor_snapshot(u32 id, struct vmm_monitor_snapshot *out)
{
	irq_flags_t flags;
	u64 now;
	struct monitor_slot *s;
	if (!out || id > CONFIG_MAX_VCPU_COUNT || !arch_atomic_read(&initialized))
		return VMM_EINVALID;
	s = &slots[id];
	vmm_spin_lock_irqsave_lite(&s->lock, flags);
	*out = s->data.snap;
	now = vmm_timer_timestamp();
	out->ready_age_ns = s->data.waiting && vmm_monitor_enabled() &&
		now >= s->data.ready_since ? now - s->data.ready_since : 0;
	vmm_spin_unlock_irqrestore_lite(&s->lock, flags);
	return out->present ? VMM_OK : VMM_ENOTAVAIL;
}

int vmm_monitor_event_get(u64 sequence, struct vmm_monitor_event *out)
{
	int rc = VMM_ENOTAVAIL;
	irq_flags_t flags;
	if (!sequence || !out)
		return VMM_EINVALID;
	vmm_spin_lock_irqsave_lite(&control_lock, flags);
	if (sequence <= status.events && status.events - sequence < VMM_MONITOR_EVENTS) {
		*out = events[(sequence - 1) % VMM_MONITOR_EVENTS];
		rc = VMM_OK;
	}
	vmm_spin_unlock_irqrestore_lite(&control_lock, flags);
	return rc;
}

int vmm_monitor_enable(bool on)
{
	u32 i;
	u64 now;
	irq_flags_t flags, sample_flags;
	if (!arch_atomic_read(&initialized))
		return VMM_ENOTAVAIL;
	vmm_spin_lock_irqsave_lite(&sample_lock, sample_flags);
	if (on == vmm_monitor_enabled())
		goto done;
	arch_atomic_write(&enabled, 0);
	if (on) {
		now = vmm_timer_timestamp();
		vmm_monitor_irq_restart(now);
		for (i = 0; i <= CONFIG_MAX_VCPU_COUNT; i++) {
			vmm_spin_lock_irqsave_lite(&slots[i].lock, flags);
			reset_data(&slots[i].data, now);
			vmm_spin_unlock_irqrestore_lite(&slots[i].lock, flags);
		}
		for (i = 0; i < CONFIG_MAX_GUEST_COUNT; i++) vmm_monitor_guest_reset(i);
		arch_wmb();
		arch_atomic_write(&enabled, 1);
	}
done:
	vmm_spin_unlock_irqrestore_lite(&sample_lock, sample_flags);
	return VMM_OK;
}

int vmm_monitor_configure(const struct vmm_monitor_config *c)
{
	u32 i;
	irq_flags_t flags, sample_flags;
	if (!c || !c->exit_rate || c->exit_rate > 1000000000 ||
	    !c->sustain_seconds || c->sustain_seconds > 60 || c->guest_cpu_limit_bp > 10000 ||
	    !arch_atomic_read(&initialized))
		return VMM_EINVALID;
	vmm_spin_lock_irqsave_lite(&sample_lock, sample_flags);
	vmm_spin_lock_irqsave_lite(&control_lock, flags);
	status.config = *c;
	vmm_spin_unlock_irqrestore_lite(&control_lock, flags);
	for (i = 0; i <= CONFIG_MAX_VCPU_COUNT; i++) {
		vmm_spin_lock_irqsave_lite(&slots[i].lock, flags);
		memset(&slots[i].data.storm, 0, sizeof(slots[i].data.storm));
		slots[i].data.previous_ns = 0;
		vmm_spin_unlock_irqrestore_lite(&slots[i].lock, flags);
	}
	vmm_spin_unlock_irqrestore_lite(&sample_lock, sample_flags);
	return VMM_OK;
}

void vmm_monitor_sample(u64 now)
{
	u32 i, alarms;
	int storm;
	u64 start = vmm_timer_timestamp(), dt;
	irq_flags_t flags, sample_flags;
	struct monitor_data *d;
	struct vmm_monitor_snapshot snap;
	struct vmm_monitor_status config;
	if (!vmm_monitor_enabled())
		return;
	if (!vmm_spin_trylock_irqsave(&sample_lock, sample_flags)) {
		vmm_spin_lock_irqsave_lite(&control_lock, flags);
		status.busy_snapshots++;
		vmm_spin_unlock_irqrestore_lite(&control_lock, flags);
		return;
	}
	if (!vmm_monitor_enabled())
		goto done;
	vmm_monitor_status(&config);
	for (i = 0; i <= CONFIG_MAX_VCPU_COUNT; i++) {
		/* Never let a contended/stuck producer stall the periodic sampler. */
		if (!vmm_spin_trylock_irqsave(&slots[i].lock, flags)) {
			vmm_spin_lock_irqsave_lite(&control_lock, flags);
			status.busy_snapshots++;
			vmm_spin_unlock_irqrestore_lite(&control_lock, flags);
			continue;
		}
		d = &slots[i].data;
		if (!d->snap.present) {
			vmm_spin_unlock_irqrestore(&slots[i].lock, flags);
			continue;
		}
		dt = now >= d->previous_ns ? now - d->previous_ns : 0;
		d->snap.valid = d->previous_ns && dt >= VMM_MONITOR_NS * 3 / 4 &&
			dt <= VMM_MONITOR_NS * 5 / 4;
		d->snap.interval_ns = d->previous_ns ? dt : 0;
		d->previous_ns = now;
		d->snap.timestamp_ns = now;
		d->snap.ready_age_ns = d->waiting && now >= d->ready_since ? now - d->ready_since : 0;
		d->snap.exits = d->exits;
		/* Divide first for absurd long gaps to keep multiplication bounded. */
		d->snap.exit_rate = dt && d->exits <= (~0ULL / VMM_MONITOR_NS) ?
			udiv64(d->exits * VMM_MONITOR_NS, dt) : 0;
		d->snap.faults = d->faults;
		vmm_monitor_hist_read(&d->ready, &d->snap.ready);
		vmm_monitor_hist_read(&d->irq, &d->snap.irq);
		vmm_monitor_hist_read(&d->dispatch, &d->snap.dispatch);
		storm = vmm_monitor_storm_step(&d->storm, d->exits, dt,
			config.config.exit_rate, config.config.sustain_seconds, d->snap.valid);
		alarms = 0;
		if (d->faults && (!d->notice_ns[0] || now - d->notice_ns[0] >= 10 * VMM_MONITOR_NS))
			alarms |= 1;
		if (config.config.ready_limit_ns &&
		    ((d->snap.ready.count && d->snap.ready.p99_upper_ns > config.config.ready_limit_ns) ||
		     d->snap.ready_age_ns > config.config.ready_limit_ns) &&
		    (!d->notice_ns[1] || now - d->notice_ns[1] >= 10 * VMM_MONITOR_NS))
			alarms |= 2;
		if (config.config.irq_limit_ns && d->snap.irq.count &&
		    d->snap.irq.p99_upper_ns > config.config.irq_limit_ns &&
		    (!d->notice_ns[2] || now - d->notice_ns[2] >= 10 * VMM_MONITOR_NS))
			alarms |= 4;
		if (alarms & 1) { d->notice_ns[0] = now; d->faults = 0; }
		if (alarms & 2) d->notice_ns[1] = now;
		if (alarms & 4) d->notice_ns[2] = now;
		d->exits = 0;
		memset(&d->ready, 0, sizeof(d->ready));
		memset(&d->irq, 0, sizeof(d->irq));
		memset(&d->dispatch, 0, sizeof(d->dispatch));
		snap = d->snap;
		vmm_spin_unlock_irqrestore(&slots[i].lock, flags);
		if (storm)
			record_event(&snap, storm > 0 ? VMM_MONITOR_EXIT_STORM :
				VMM_MONITOR_EXIT_RECOVER, snap.exit_rate, config.config.exit_rate, snap.last_exit);
		if (alarms & 1)
			record_event(&snap, VMM_MONITOR_MEMORY, snap.faults, snap.last_address, snap.last_fault);
		if (alarms & 2)
			record_event(&snap, VMM_MONITOR_READY_SLOW,
				snap.ready_age_ns > snap.ready.p99_upper_ns ? snap.ready_age_ns : snap.ready.p99_upper_ns,
				config.config.ready_limit_ns, snap.ready_age_ns > snap.ready.p99_upper_ns);
		if (alarms & 4)
			record_event(&snap, VMM_MONITOR_IRQ_SLOW, snap.irq.p99_upper_ns, config.config.irq_limit_ns, 0);
	}
	vmm_monitor_guest_sample(now, config.config.guest_cpu_limit_bp);
	vmm_monitor_platform_sample(now);
	dt = vmm_timer_timestamp() - start;
	vmm_spin_lock_irqsave_lite(&control_lock, flags);
	status.samples++;
	if (dt > status.sample_max_ns) status.sample_max_ns = dt;
	vmm_spin_unlock_irqrestore_lite(&control_lock, flags);
done:
	vmm_spin_unlock_irqrestore_lite(&sample_lock, sample_flags);
}

/* Sole worker; all formatting and potentially slow output stays here. */
void vmm_monitor_drain(void)
{
	u32 n;
	int len, rc;
	irq_flags_t flags;
	char json[384];
	struct vmm_monitor_event e;
	struct vmm_monitor_status c;
	if (!arch_atomic_read(&initialized))
		return;
	vmm_mutex_lock(&drain_lock);
	vmm_monitor_status(&c);
	for (n = 0; n < VMM_MONITOR_BATCH; n++) {
		vmm_spin_lock_irqsave_lite(&control_lock, flags);
		if (report_sequence > status.events) {
			vmm_spin_unlock_irqrestore_lite(&control_lock, flags);
			break;
		}
		e = events[(report_sequence++ - 1) % VMM_MONITOR_EVENTS];
		vmm_spin_unlock_irqrestore_lite(&control_lock, flags);
		len = vmm_snprintf(json, sizeof(json),
			"{\"schema\":2,\"seq\":%llu,\"time_ns\":%llu,\"type\":%u,"
			"\"guest\":%u,\"vcpu\":%u,\"generation\":%u,\"value\":%llu,"
			"\"detail\":%llu,\"code\":%u}\n",
			(unsigned long long)e.sequence, (unsigned long long)e.timestamp_ns,
			e.type, e.guest_id, e.vcpu_id, e.generation,
			(unsigned long long)e.value, (unsigned long long)e.detail, e.code);
		if (len < 0 || len >= sizeof(json))
			continue;
		if (c.config.log)
			vmm_printf("%s", json);
		if (c.config.publish) {
			rc = vmm_monitor_platform_publish(json, len);
			if (rc) {
				vmm_spin_lock_irqsave_lite(&control_lock, flags);
				status.publish_errors++;
				vmm_spin_unlock_irqrestore_lite(&control_lock, flags);
			}
		}
	}
	vmm_mutex_unlock(&drain_lock);
}

#ifndef CONFIG_MONITOR_EXTERNAL_EXECUTOR
static void report_fn(struct vmm_work *work) { vmm_monitor_drain(); }
/* IRQ context: no mutex, allocation, formatting, or Guest calls. */
static void monitor_timer(struct vmm_timer_event *ev)
{
	u64 now = vmm_timer_timestamp();
	irq_flags_t flags;
	int rc;
	if (arch_atomic_read(&executor_state) != 1) {
		if (!arch_atomic_read(&executor_state))
			vmm_timer_event_start(ev, VMM_MONITOR_NS);
		return;
	}
	if (vmm_monitor_enabled() && now > next_sample && now - next_sample >= VMM_MONITOR_NS / 4) {
		vmm_spin_lock_irqsave_lite(&control_lock, flags);
		status.late_samples++;
		vmm_spin_unlock_irqrestore_lite(&control_lock, flags);
	}
	vmm_monitor_sample(now);
	rc = vmm_workqueue_schedule_work(report_queue, &report_work);
	if (rc && rc != VMM_EALREADY) {
		vmm_spin_lock_irqsave_lite(&control_lock, flags);
		status.work_errors++;
		vmm_spin_unlock_irqrestore_lite(&control_lock, flags);
	}
	now = vmm_timer_timestamp();
	next_sample += (udiv64(now >= next_sample ? now - next_sample : 0, VMM_MONITOR_NS) + 1) * VMM_MONITOR_NS;
	if (vmm_timer_event_start(ev, next_sample - now)) {
		vmm_spin_lock_irqsave_lite(&control_lock, flags);
		status.work_errors++;
		vmm_spin_unlock_irqrestore_lite(&control_lock, flags);
	}
}

static void monitor_timer_start(void *unused0, void *unused1, void *unused2)
{
	/* Static result avoids a stack pointer escaping a timed-out IPI call. */
	if (arch_atomic_read(&executor_state) < 0) return;
	next_sample = vmm_timer_timestamp() + VMM_MONITOR_NS;
	arch_atomic_write(&timer_start_result, vmm_timer_event_start(&sample_timer, VMM_MONITOR_NS));
}
#endif

int __init vmm_monitor_init(void)
{
	u32 i;
#ifndef CONFIG_MONITOR_EXTERNAL_EXECUTOR
	int rc;
	u64 init_deadline;
#endif
	for (i = 0; i <= CONFIG_MAX_VCPU_COUNT; i++)
		INIT_SPIN_LOCK(&slots[i].lock);
	vmm_monitor_irq_init();
	INIT_SPIN_LOCK(&sample_lock);
	vmm_monitor_guest_init();
	vmm_monitor_diag_init();
	INIT_MUTEX(&drain_lock);
	status.config.exit_rate = 50000;
	status.config.sustain_seconds = 5;
	status.config.log = TRUE;
	status.config.guest_cpu_limit_bp = 9000;
#ifndef CONFIG_MONITOR_EXTERNAL_EXECUTOR
	INIT_TIMER_EVENT(&sample_timer, monitor_timer, NULL);
	INIT_WORK(&report_work, report_fn);
	if (CONFIG_MONITOR_HCPU >= CONFIG_CPU_COUNT ||
	    !vmm_cpu_online(CONFIG_MONITOR_HCPU))
		return VMM_ENOTAVAIL;
	report_queue = vmm_workqueue_create("monitor_wq", CONFIG_MONITOR_PRIORITY);
	if (!report_queue)
		return VMM_ENOMEM;
	rc = vmm_threads_set_affinity(vmm_workqueue_get_thread(report_queue),
				      vmm_cpumask_of(CONFIG_MONITOR_HCPU));
	if (rc)
		goto fail_queue;
	/* Timer starts on the configured core, independently of vCPU priorities. */
	rc = vmm_smp_ipi_sync_call(vmm_cpumask_of(CONFIG_MONITOR_HCPU), 1000,
				   monitor_timer_start, NULL, NULL, NULL);
	if (rc) goto fail_queue;
	/* Upstream sync IPI waits for dequeue, not necessarily callback completion. */
	init_deadline = vmm_timer_timestamp() + VMM_MONITOR_NS;
	while (arch_atomic_read(&timer_start_result) == VMM_EINPROGRESS &&
	       vmm_timer_timestamp() < init_deadline)
		vmm_udelay(1);
	rc = arch_atomic_read(&timer_start_result);
	if (rc == VMM_EINPROGRESS) rc = VMM_ETIMEDOUT;
	if (rc) goto fail_queue;
#endif
	arch_wmb();
	arch_atomic_write(&initialized, 1);
#ifndef CONFIG_MONITOR_EXTERNAL_EXECUTOR
	arch_atomic_write(&executor_state, 1);
#endif
#ifdef CONFIG_MONITOR_START_ENABLED
	vmm_monitor_enable(TRUE);
#endif
	return VMM_OK;
#ifndef CONFIG_MONITOR_EXTERNAL_EXECUTOR
fail_queue:
	arch_atomic_write(&executor_state, -1);
	vmm_workqueue_destroy(report_queue);
	return rc;
#endif
}

#include <vmm_monitor_diag.h>
#include <vmm_manager.h>
#include <vmm_timer.h>
#include <libs/stringlib.h>

static DEFINE_SPINLOCK(diag_lock);
static atomic_t armed = {0};
static u64 begin_ns, end_ns, overflow;
static struct vmm_monitor_diag_record records[VMM_MONITOR_DIAG_SLOTS];

void vmm_monitor_diag_init(void) { INIT_SPIN_LOCK(&diag_lock); }

int vmm_monitor_diag_start(u32 ms)
{
	irq_flags_t flags;
	if (!ms || ms > 5000) return VMM_EINVALID;
	vmm_spin_lock_irqsave_lite(&diag_lock, flags);
	if (arch_atomic_read(&armed)) {
		vmm_spin_unlock_irqrestore_lite(&diag_lock, flags);
		return VMM_EBUSY;
	}
	memset(records, 0, sizeof(records)); overflow = 0;
	begin_ns = vmm_timer_timestamp(); end_ns = begin_ns + (u64)ms * 1000000;
	arch_atomic_write(&armed, 1);
	vmm_spin_unlock_irqrestore_lite(&diag_lock, flags);
	return VMM_OK;
}

u64 vmm_monitor_diag_begin(void)
{
	return arch_atomic_read(&armed) ? vmm_timer_timestamp() : 0;
}

void vmm_monitor_diag_end(struct vmm_vcpu *v, u64 pc, u64 esr, u64 start, int rc)
{
	u32 i;
	u64 now, ns;
	irq_flags_t flags;
	struct vmm_monitor_diag_record *r;
	if (!start || !arch_atomic_read(&armed) || !v || !v->is_normal) return;
	now = vmm_timer_timestamp();
	vmm_spin_lock_irqsave_lite(&diag_lock, flags);
	if (!arch_atomic_read(&armed) || start < begin_ns || now < start || now > end_ns)
		goto done;
	ns = now - start;
	for (i = 0; i < VMM_MONITOR_DIAG_SLOTS; i++) {
		r = &records[i];
		if (!r->count || (r->guest == v->guest->id && r->vcpu == v->id &&
		    r->pc == pc && (r->esr >> 26) == (esr >> 26))) {
			r->guest = v->guest->id; r->vcpu = v->id;
			r->pc = pc; r->esr = esr; r->last_ns = now;
			r->count++; r->sum_ns += ns; if (ns > r->max_ns) r->max_ns = ns;
			if (rc) r->errors++;
			goto done;
		}
	}
	overflow++;
done:
	vmm_spin_unlock_irqrestore_lite(&diag_lock, flags);
}

void vmm_monitor_diag_stop(void)
{
	irq_flags_t flags;
	vmm_spin_lock_irqsave_lite(&diag_lock, flags);
	arch_atomic_write(&armed, 0);
	vmm_spin_unlock_irqrestore_lite(&diag_lock, flags);
}

int vmm_monitor_diag_get(u32 index, struct vmm_monitor_diag_record *out)
{
	irq_flags_t flags;
	if (!out || index >= VMM_MONITOR_DIAG_SLOTS) return VMM_EINVALID;
	vmm_spin_lock_irqsave_lite(&diag_lock, flags);
	*out = records[index];
	vmm_spin_unlock_irqrestore_lite(&diag_lock, flags);
	return out->count ? VMM_OK : VMM_ENOTAVAIL;
}

u64 vmm_monitor_diag_overflow(void)
{
	u64 count;
	irq_flags_t flags;
	vmm_spin_lock_irqsave_lite(&diag_lock, flags);
	count = overflow;
	vmm_spin_unlock_irqrestore_lite(&diag_lock, flags);
	return count;
}

	vmm_spinlock_t lock;
	struct vmm_monitor_guest report;
	u64 last_notice, last_sample_sequence;
} guests[CONFIG_MAX_GUEST_COUNT];

void vmm_monitor_guest_init(void)
{
	u32 i;
	for (i = 0; i < CONFIG_MAX_GUEST_COUNT; i++)
		INIT_SPIN_LOCK(&guests[i].lock);
}

void vmm_monitor_guest_reset(u32 id)
{
	irq_flags_t flags;
	if (id >= CONFIG_MAX_GUEST_COUNT) return;
	vmm_spin_lock_irqsave_lite(&guests[id].lock, flags);
	memset(&guests[id].report, 0, sizeof(guests[id].report));
	guests[id].last_notice = guests[id].last_sample_sequence = 0;
	vmm_spin_unlock_irqrestore_lite(&guests[id].lock, flags);
}

int vmm_monitor_guest_report(struct vmm_vcpu *actor, u64 session, u64 sequence,
		u64 cpu_bp, u64 total, u64 available, u64 free_bytes, u64 valid)
{
	u32 id;
	int rc = VMM_EINVALID;
	irq_flags_t flags;
	struct vmm_monitor_guest *r;
	/* Identity comes only from the trapping vCPU, never from Guest input. */
	if (!actor || !actor->is_normal || !actor->guest ||
	    actor->guest->id >= CONFIG_MAX_GUEST_COUNT || !session || !sequence ||
	    !valid || valid > 3 || cpu_bp > 10000 || available > total ||
	    free_bytes > total || ((valid & 2) && !total))
		return VMM_EINVALID;
	if (!vmm_monitor_enabled()) return VMM_ENOTAVAIL;
	id = actor->guest->id;
	vmm_spin_lock_irqsave_lite(&guests[id].lock, flags);
	r = &guests[id].report;
	if (r->session != session || sequence > r->sequence) {
		if (r->session != session) {
			guests[id].last_notice = 0;
			guests[id].last_sample_sequence = 0;
		}
		r->session = session; r->sequence = sequence;
		r->received_ns = vmm_timer_timestamp();
		r->total_bytes = total; r->available_bytes = available;
		r->free_bytes = free_bytes; r->cpu_bp = cpu_bp; r->valid = valid;
		rc = VMM_OK;
	}
	vmm_spin_unlock_irqrestore_lite(&guests[id].lock, flags);
	return rc;
}

int vmm_monitor_guest_get(u32 id, struct vmm_monitor_guest *out)
{
	irq_flags_t flags;
	if (id >= CONFIG_MAX_GUEST_COUNT || !out) return VMM_EINVALID;
	vmm_spin_lock_irqsave_lite(&guests[id].lock, flags);
	*out = guests[id].report;
	vmm_spin_unlock_irqrestore_lite(&guests[id].lock, flags);
	return out->sequence ? VMM_OK : VMM_ENOTAVAIL;
}

void vmm_monitor_guest_sample(u64 now, u32 limit_bp)
{
	u32 i, value;
	bool alarm;
	irq_flags_t flags;
	struct vmm_monitor_guest *r;
	for (i = 0; i < CONFIG_MAX_GUEST_COUNT; i++) {
		if (!vmm_spin_trylock_irqsave(&guests[i].lock, flags)) continue;
		r = &guests[i].report;
		alarm = FALSE; value = r->cpu_bp;
		if (r->sequence != guests[i].last_sample_sequence && now >= r->received_ns &&
		    now - r->received_ns <= 3 * VMM_MONITOR_NS) {
			guests[i].last_sample_sequence = r->sequence;
			if ((r->valid & 1) && limit_bp && value > limit_bp &&
			    (!guests[i].last_notice || now - guests[i].last_notice >= 10 * VMM_MONITOR_NS)) {
				guests[i].last_notice = now; alarm = TRUE;
			}
		}
		vmm_spin_unlock_irqrestore(&guests[i].lock, flags);
		if (alarm) vmm_monitor_guest_event(i, now, value, limit_bp);
	}
}

#include <vmm_monitor.h>

#include <vmm_timer.h>

/* One slot per outstanding physical-to-virtual delivery, not per interrupt.
 * Static storage: no changes to the allocation size of Guest/VGIC objects.
 * Caller holds VGIC dist_lock for pending/injected/reset routing operations.
 */
struct pending_irq { u64 start; u32 virq; };
static struct {
	vmm_spinlock_t lock;
	u64 epoch_ns;
	struct pending_irq irq[CONFIG_MONITOR_IRQ_SLOTS];
} pending[CONFIG_MAX_VCPU_COUNT];
static struct vmm_monitor_hirq_context current[CONFIG_CPU_COUNT];
static atomic64_t untracked = { 0 };

/* Called once from monitor init, before collection can be enabled. */
void vmm_monitor_irq_init(void)
{
	u32 i;
	for (i = 0; i < CONFIG_MAX_VCPU_COUNT; i++)
		INIT_SPIN_LOCK(&pending[i].lock);
}

void vmm_monitor_irq_restart(u64 now)
{
	u32 v, i;
	irq_flags_t flags;
	for (v = 0; v < CONFIG_MAX_VCPU_COUNT; v++) {
		vmm_spin_lock_irqsave_lite(&pending[v].lock, flags);
		pending[v].epoch_ns = now;
		for (i = 0; i < CONFIG_MONITOR_IRQ_SLOTS; i++)
			pending[v].irq[i].start = 0;
		vmm_spin_unlock_irqrestore_lite(&pending[v].lock, flags);
	}
}

struct vmm_monitor_hirq_context vmm_monitor_hirq_enter(u32 hirq)
{
	u32 cpu = vmm_smp_processor_id();
	struct vmm_monitor_hirq_context old = current[cpu];
	current[cpu].timestamp_ns = vmm_monitor_enabled() ? vmm_timer_timestamp() : 0;
	current[cpu].hirq = hirq;
	return old;
}

void vmm_monitor_hirq_exit(struct vmm_monitor_hirq_context old)
{
	current[vmm_smp_processor_id()] = old;
}

void vmm_monitor_irq_reset(struct vmm_vcpu *v)
{
	u32 i;
	irq_flags_t flags;
	if (!v || !v->is_normal || v->id >= CONFIG_MAX_VCPU_COUNT)
		return;
	vmm_spin_lock_irqsave_lite(&pending[v->id].lock, flags);
	pending[v->id].epoch_ns = vmm_timer_timestamp();
	for (i = 0; i < CONFIG_MONITOR_IRQ_SLOTS; i++)
		pending[v->id].irq[i].start = 0;
	vmm_spin_unlock_irqrestore_lite(&pending[v->id].lock, flags);
}

void vmm_monitor_irq_pending(struct vmm_vcpu *v, u32 virq, u32 hirq)
{
	struct vmm_monitor_hirq_context c;
	if (!vmm_monitor_enabled() || !v || !v->is_normal ||
	    v->id >= CONFIG_MAX_VCPU_COUNT || hirq == VMM_MONITOR_UNKNOWN)
		return;
	c = current[vmm_smp_processor_id()];
	/* Do not pair an unrelated interrupt or a software-generated assertion. */
	if (!c.timestamp_ns || c.hirq != hirq) {
		arch_atomic64_inc(&untracked);
		return;
	}
	vmm_monitor_irq_pending_at(v, virq, c.timestamp_ns);
}

void vmm_monitor_irq_pending_at(struct vmm_vcpu *v, u32 virq, u64 start)
{
	u32 i, free_slot = CONFIG_MONITOR_IRQ_SLOTS;
	irq_flags_t flags;
	struct pending_irq *p;
	if (!vmm_monitor_enabled() || !v || !v->is_normal ||
	    v->id >= CONFIG_MAX_VCPU_COUNT || !start) return;
	vmm_spin_lock_irqsave_lite(&pending[v->id].lock, flags);
	if (!vmm_monitor_enabled() || start < pending[v->id].epoch_ns)
		goto done;
	p = pending[v->id].irq;
	for (i = 0; i < CONFIG_MONITOR_IRQ_SLOTS; i++) {
		if (p[i].start && p[i].virq == virq)
			goto done; /* Coalesced level: preserve earliest receipt. */
		if (!p[i].start)
			free_slot = i;
	}
	if (free_slot < CONFIG_MONITOR_IRQ_SLOTS) {
		p[free_slot].virq = virq;
		p[free_slot].start = start;
	} else {
		arch_atomic64_inc(&untracked);
	}
done:
	vmm_spin_unlock_irqrestore_lite(&pending[v->id].lock, flags);
}

/* Caller serializes the real IRQ route/MOVI with pending/injection operations. */
void vmm_monitor_irq_move(struct vmm_vcpu *from, struct vmm_vcpu *to, u32 virq)
{
	u32 first, second, i, free_slot = CONFIG_MONITOR_IRQ_SLOTS;
	u64 start = 0;
	irq_flags_t f1, f2;
	struct pending_irq *p;
	if (!vmm_monitor_enabled() || !from || !to || !from->is_normal || !to->is_normal ||
	    from->id >= CONFIG_MAX_VCPU_COUNT || to->id >= CONFIG_MAX_VCPU_COUNT ||
	    from->id == to->id || from->guest != to->guest) return;
	first = from->id < to->id ? from->id : to->id;
	second = from->id < to->id ? to->id : from->id;
	vmm_spin_lock_irqsave_lite(&pending[first].lock, f1);
	vmm_spin_lock_irqsave_lite(&pending[second].lock, f2);
	for (i = 0; i < CONFIG_MONITOR_IRQ_SLOTS; i++)
		if (pending[from->id].irq[i].start && pending[from->id].irq[i].virq == virq) {
			start = pending[from->id].irq[i].start; pending[from->id].irq[i].start = 0; break;
		}
	if (!start) goto done;
	if (start < pending[to->id].epoch_ns) goto lost;
	p = pending[to->id].irq;
	for (i = 0; i < CONFIG_MONITOR_IRQ_SLOTS; i++) {
		if (p[i].start && p[i].virq == virq) {
			if (start < p[i].start) p[i].start = start;
			goto done;
		}
		if (!p[i].start) free_slot = i;
	}
	if (free_slot < CONFIG_MONITOR_IRQ_SLOTS) {
		p[free_slot].virq = virq; p[free_slot].start = start; goto done;
	}
lost:
	arch_atomic64_inc(&untracked);
done:
	vmm_spin_unlock_irqrestore_lite(&pending[second].lock, f2);
	vmm_spin_unlock_irqrestore_lite(&pending[first].lock, f1);
}

static void irq_finish(struct vmm_vcpu *v, u32 virq, bool injected)
{
	u32 i;
	u64 start = 0, end;
	irq_flags_t flags;
	if (!vmm_monitor_enabled() || !v || !v->is_normal || v->id >= CONFIG_MAX_VCPU_COUNT)
		return;
	end = injected ? vmm_timer_timestamp() : 0;
	vmm_spin_lock_irqsave_lite(&pending[v->id].lock, flags);
	for (i = 0; i < CONFIG_MONITOR_IRQ_SLOTS; i++) {
		if (pending[v->id].irq[i].start && pending[v->id].irq[i].virq == virq) {
			start = pending[v->id].irq[i].start;
			pending[v->id].irq[i].start = 0;
			break;
		}
	}
	vmm_spin_unlock_irqrestore_lite(&pending[v->id].lock, flags);
	if (start && end)
		vmm_monitor_irq_sample(v, start, end);
}

void vmm_monitor_irq_injected(struct vmm_vcpu *v, u32 virq)
{
	irq_finish(v, virq, TRUE);
}

void vmm_monitor_irq_cancel(struct vmm_vcpu *v, u32 virq)
{
	irq_finish(v, virq, FALSE);
}

u64 vmm_monitor_irq_untracked(void)
{
	return arch_atomic64_read(&untracked);
}

#include <vmm_monitor.h>
#include <libs/mathlib.h>

void vmm_monitor_hist_add(struct vmm_monitor_hist *h, u64 ns)
{
	u32 b = 0;
	u64 upper = 1000;
	while (b < VMM_MONITOR_BUCKETS - 1 && ns > upper) {
		upper <<= 1;
		b++;
	}
	if (!h->count || ns < h->min)
		h->min = ns;
	if (ns > h->max)
		h->max = ns;
	h->count++;
	h->sum += ns;
	h->bins[b]++;
}

void vmm_monitor_hist_read(const struct vmm_monitor_hist *h,
			   struct vmm_monitor_latency *out)
{
	u32 b;
	u64 n = 0, target = h->count - udiv64(h->count, 100);
	out->count = h->count;
	out->min_ns = h->min;
	out->max_ns = h->max;
	out->mean_ns = h->count ? udiv64(h->sum, h->count) : 0;
	out->p99_upper_ns = 0;
	if (!h->count)
		return;
	for (b = 0; b < VMM_MONITOR_BUCKETS; b++) {
		n += h->bins[b];
		if (n >= target) {
			/* Last bucket is unbounded; observed maximum is a safe bound. */
			out->p99_upper_ns = b == VMM_MONITOR_BUCKETS - 1 ?
				h->max : (1000ULL << b);
			return;
		}
	}
}

int vmm_monitor_storm_step(struct vmm_monitor_streak *s, u64 count,
			   u64 dt, u32 rate, u32 seconds, bool valid)
{
	/* dt is bounded before multiplying; do not credit missing windows. */
	if (!valid || dt < VMM_MONITOR_NS * 3 / 4 ||
	    dt > VMM_MONITOR_NS * 5 / 4) {
		s->elapsed = 0;
		s->windows = 0;
		return 0;
	}
	/* count > rate * dt / 1e9 implements strict 'greater than'. */
	if (count > udiv64((u64)rate * dt, VMM_MONITOR_NS)) {
		if (s->elapsed < (u64)seconds * VMM_MONITOR_NS)
			s->elapsed += dt;
		if (s->windows < seconds)
			s->windows++;
		if (!s->active && s->windows >= seconds &&
		    s->elapsed >= (u64)seconds * VMM_MONITOR_NS) {
			s->active = TRUE;
			return 1;
		}
	} else {
		bool was_active = s->active;
		s->elapsed = 0;
		s->windows = 0;
		s->active = FALSE;
		return was_active ? -1 : 0;
	}
	return 0;
}

static bool shmem_guest_allowed(struct vmm_vcpu *actor, struct vmm_shmem *shm,
	physical_addr_t gpa, physical_addr_t off, u32 len, bool write)
{
	physical_addr_t hpa;
	physical_size_t available;
	u32 flags;
	struct vmm_region *region;
	if (!actor || !actor->is_normal || !shm) return FALSE;
	if (!vmm_monitor_range_valid(shm->size, off, len) ||
	    !vmm_monitor_range_valid(~0ULL, shm->addr, off) ||
	    !vmm_monitor_range_valid(~0ULL, shm->addr + off, len) ||
	    !vmm_monitor_range_valid(~0ULL, gpa, len)) {
		vmm_monitor_mem_fault(actor, VMM_MONITOR_SHMEM_BOUNDS, gpa, len);
		return FALSE;
	}
	region = vmm_guest_find_region(actor->guest, gpa, VMM_REGION_MEMORY, FALSE);
	if (!region || (region->flags & VMM_REGION_ALIAS) ||
	    vmm_guest_physical_map(actor->guest, gpa, len, &hpa, &available, &flags) ||
	    available < len || hpa != shm->addr + off || !(flags & VMM_REGION_REAL) ||
	    !(flags & (VMM_REGION_ISRAM | VMM_REGION_ISSHARED)) ||
	    (write && (flags & VMM_REGION_READONLY))) {
		vmm_monitor_mem_fault(actor, VMM_MONITOR_ACCESS_DENIED, gpa, write);
		return FALSE;
	}
	return TRUE;
}

# monitor_agent 第二版：实现、合入与验收说明

日期：2026-10-21。基于当前 Xvisor 开源参考树，ARM64、12 核；产品 auth=4、mterm=3。本版的日志工作队列默认优先级为 3，采样改为定时器 IRQ。**这不是全部指标已经验收通过的声明：5 KB ROM 仍未达标，目标板 CPU <0.5% 未验证，精确 PMU 周期数未实现。**

## 1. 对修改后的需求逐项回答

“已有代码”与“产品路径已接入、板端验收通过”分别记录。新的要求只要求 exit 风暴触发日志，不再要求死机现场；当前结构化风暴/恢复日志可以对应这一功能。

## 2. 执行模型

```mermaid
flowchart LR
    S[调度器及异常打点] --> A[静态 vCPU 统计]
    P[物理 IRQ 分发] --> I[有界 IRQ 关联表]
    I --> A
    T[HV 定时器 IRQ 每秒] --> A
    T --> E[固定事件环]
    T --> W[后台日志工作队列 优先级3]
    E --> W
    W --> J[JSON 日志]
    G[Linux或Android采集程序] --> B[Guest内核HVC桥]
    B --> R[HV按真实调用Guest接收]
    R --> T
    C[monitor命令] --> A
    C --> R
    C --> D[按需指令陷入诊断]
```

内部模式只创建一个专用输出工作队列，没有优先级 6 的常驻采样线程。静态 timer 在 `CONFIG_MONITOR_HCPU` 对应核启动，默认 0；队列使用同一 affinity、优先级默认 3。`CONFIG_MONITOR_PRIORITY` 现在**只控制日志线程**。不要因 mterm/auth 偶现问题重新把采样搬回前台忙轮询。

`sample_lock` 已由 mutex 改为自旋控制锁，采样只尝试取得它；拿不到就记录 busy，不等待配置路径。单槽也 trylock；统计不全时按实际 dt 标质量，不制造完整窗口。生产者、控制和事件仍使用短锁，不能将 trylock 描述成整个系统完全无等待。

`vmm_monitor_platform_sample()` 现在处于 IRQ 上下文：产品覆盖函数必须无等待、无分配、无日志、无递归 sample/enable/configure。`vmm_monitor_drain()` 仍使用 mutex，仅可后台调用。发布函数也仅在后台调用。

初始化为 IPI 延迟/超时和 timer 启动失败设置静态状态；失败后的晚到回调不访问已经销毁的队列。配置核必须在线；CPU 热拔出不在本产品静态 12 核设计中，若产品支持 hotplug 需增加迁移/停用流程。

stop 后不新增周期指标及 Guest 遥测；历史事件仍可排出。再次 start 清监控槽、IRQ 起点和 Guest 遥测缓存，下一轮预热。诊断命令是独立按需能力，允许在周期 monitor 停止时使用。

## 3. CPU 时间口径和打点

| 指标 | 起点 | 终点 | 不包含 |
|---|---|---|---|
| ready | 状态真正转为 READY 的通知 | 选出且已验证目标 vCPU，准备安装上下文的 selected 通知 | selected 后的上下文切换耗时；进入 READY 前的唤醒处理 |
| dispatch | `__vmm_scheduler_next1/next2` 入口原有 tstamp | arch_vcpu_switch 完成后打点；同一 vCPU 再选中也记录 | Guest 实际执行；调度函数后续记账尾部 |
| host_irq_to_lr | 确认有效物理 IRQ 后的分发软件时间 | 成功 set_lr 后 | 硬件边沿到软件入口、Guest ISR 开始及执行 |

`dispatch` 是该次调度路径延迟，包含出队、必要的旧状态处理、锁和上下文安装；不是仅测汇编切换，也不是从唤醒请求到 Guest 第一条指令。两个区间可以重叠，不能简单相加得出总延迟。

`vmm_monitor_selected()` 在目标 sched_lock 保护下记录 READY 样本并清 waiting。`vmm_monitor_state(RUNNING)` 保留兼容兜底：遗漏 selected 的旧集成仍能得到 READY→RUNNING 样本，但这不符合新版精确口径，产品验收须确认 selected 已合入。

直方图仍为 16 桶；P99 是桶上界，允许大于实测最大值；零 count 是无样本。PAUSE/HALT/RESET/UNKNOWN 撤销未完成等待。一直 READY 时仍通过 ready_age_ns 检测饥饿。

由于 ready 口径变化，HV JSON **schema 升为 2**；旧 QEMU schema=1 数据不被重解释。新版解析脚本兼容读入 1 和 2，并保留 schema。

## 4. 一键指令陷入诊断

```text
monitor diagnose
monitor diagnose 2000
monitor collect
```

diagnose 默认观察 1000 ms，允许 1～5000 ms；命令阻塞等待但采用 sleep，不忙轮询。collect 输出缓存后执行一次 1000 ms 诊断。命令 mutex 防止两次诊断互相清空数据。

ARM64 `do_sync()` 排除 EL2 自身异常后开始计时，仿真处理完成、原有错误日志之前结束；保存进入时的 ELR/ESR，避免处理函数修改 regs 后丢失位置。静态表保存最先遇到的最多 16 个 `(guest, vcpu, ELR, EC)` 组合，重复项累计；容量不足记 `untracked`，它不是精确 Top-N。

诊断行包含：guest/vcpu、elr、esr、ec、count、mean_service_ns、max_service_ns、last_ns、errors、suspect。suspect 位：1=同位置速率严格超过 1000 次/秒；2=最大服务经过时间 >1 ms；4=异常处理返回错误。WFI/WFE 不参与位 1/2 判断，避免把正常等待当重复指令异常；错误位仍保留。

这些是疑似条件而非故障定论；经过时间包含该路径上的等待/抢占，不等于 CPU 执行周期。HVC/SMC 的 ELR 可能是返回地址，定位 A64 调用指令时结合 EC 和反汇编处理，不能把 ELR 统一当成 fault 指令地址。

内置诊断汇总明确输出 `pmu_cycles="not_collected"`。随后保留调用 `vmm_monitor_platform_diagnose()`，产品覆盖该接口时应输出独立记录；默认返回不支持，不影响内置诊断成功。没有修改 PMCR/PMU 计数器，避免占用 Guest 或平台已有 PMU。要实现真正 cycles/instructions/CPI，必须先确定 PMU 所有权、虚拟化方式和过滤口径；本包未将服务耗时换算成伪造的 CPU 周期。

## 5. Guest CPU/内存采集与 HV 查询

### 5.1 Guest 端构建和部署

新增目录 `tools/monitor_guest/`：

- `monitor_guest.c`、`monitor_parse.h`：Linux/Android 用户态采集程序。
- `xvisor_monitor_bridge.c`、`monitor_uapi.h`：ARM64 Guest 内核设备桥。
- `Makefile`：用户程序和内核模块构建入口。
- `parse_monitor_jsonl.py`：HV 输出解析示例。

Linux 示例（使用 Guest 对应的内核构建树和工具链）：

```sh
make -C tools/monitor_guest CC=aarch64-linux-gnu-gcc
make -C tools/monitor_guest bridge KDIR=/path/to/guest-kernel-build ARCH=arm64 CROSS_COMPILE=aarch64-linux-gnu-
# 将程序和按产品策略签名/集成的驱动放入 Guest
insmod xvisor_monitor_bridge.ko
./monitor_guest --interval-ms 1000 --cpu-limit-bp 9000
```

Android 用户程序使用对应 NDK 的 AArch64 clang 构建；桥按 Android 产品内核构建，模块或内建均需符合项目的内核配置和签名方式。采集服务需访问 `/proc/stat`、`/proc/meminfo` 和 `/dev/xvisor_monitor`，并具有 CAP_SYS_ADMIN；设备默认 0600，SELinux 权限由产品配置。

**仅在已合入本包 HVC 分支的 ARM64 Guest 环境运行上报。**旧固件会把私有 HVC 交给原处理路径；可能注入未定义指令，不能盲测。驱动加载本身不执行 HVC，首次 write 才发送。只查看 Guest 统计时可使用：

```sh
./monitor_guest --stdout-only
./monitor_guest --stdout-only --once
```

采集周期允许 100～2000 ms，默认 1000 ms。CPU 使用 `/proc/stat` 前八个汇总字段的增量；guest/guest_nice 已包括在原 CPU 字段中，不再累加。idle+iowait 计入空闲；计数倒退、总增量为零时 CPU valid 位不置位，重新建立基线。

内存采集 `MemTotal`、`MemAvailable`、`MemFree`，kB 按 1024 转成字节；缺少字段、溢出或容量关系异常时内存 valid 位不置位。`used_estimate_bytes=total-available` 是约定估算口径，不等于所有进程 RSS/PSS 相加。CMA 不在本次新增需求中，没有额外实现。

Guest 每期输出 JSON，其中 `cpu_high` 表示超过 Guest 命令门限。HV 独立门限使用 `monitor config cpu_bp 9000`；10000=100%，9000=90%，0=关闭该告警。部署时同步设置两端门限。

### 5.2 私有 HVC ABI

| 位置 | 值/含义 |
|---|---|
| 指令 | `hvc #0x4d4f`，与 PSCI 常用 `hvc #0` 分离 |
| x0 输入 | `0x584d4f4e00000001`，协议 magic/version |
| x1 | 驱动本次加载会话随机非零值 |
| x2 | 本会话递增序号，普通报告非零；0 仅版本查询 |
| x3 | CPU basis points，0～10000 |
| x4/x5/x6 | total/available/free，字节 |
| x7 | 有效位：bit0=CPU，bit1=内存；至少一项有效 |
| x0 输出 | 报告成功为 0，失败为 VMM 错误；版本查询返回 1 |

HV 仅拦截匹配的 A64 HVC EC、immediate 和 magic；其他 HVC 仍走原路径。HVC 的返回 PC 不额外加 4。身份取自实际陷入的 `vcpu->guest`，报文没有可伪造的目标 Guest ID，也没有 Guest 指针/物理地址参数。

HV 校验字段范围、valid 位和同会话序号；每 Guest 固定一个槽，原子替换报告。不同会话允许重新从序号 1 开始；这不是对恶意 Guest 自报告的真实性认证。每个 Guest 只能更新自己的槽。

### 5.3 HV 命令和质量

```text
monitor start
monitor guest
monitor cpu
monitor mem Linux
monitor config cpu_bp 9000
monitor events
monitor stop
```

`monitor guest` 输出每个存在 VM 的 OS 报告；`monitor cpu` 先输出 OS 报告，再输出 vCPU 时延快照。`monitor mem <name>` 输出指定 Guest 报告及区域表。当前开源树仍使用 `guest region_list`；产品改成 `vm region_list` 时保留这一适配，并核对 Guest 查找 API 是否也改名。

OS 报告质量：missing=没有报告；disabled=HV monitor 关闭；stale=距 HV 接收时间超过 3 秒或时间异常；valid=接收时间有效。另必须检查 valid 位决定 CPU/内存字段是否可用。**valid 不是 Guest 健康证明。**

HV 接收时间使用 HV 单调时钟，不和 Guest 时钟直接相减。报告包含 session/seq/received_ns/age_ns；Guest 复位、监控重开清缓存。CPU 高负载事件 type=6，value=cpu_bp，detail=HV 门限，vcpu=UNKNOWN；只用新报告触发，每 Guest 最多 10 秒一条。

## 6. 内存保护与私有路径接入

原 stage-2 和 shmem 埋点保留。新增：

```c
u32 vmm_shmem_guest_read(struct vmm_vcpu *actor, struct vmm_shmem *shm,
    physical_addr_t gpa, physical_addr_t off, void *dst, u32 len, bool cacheable);
u32 vmm_shmem_guest_write(struct vmm_vcpu *actor, struct vmm_shmem *shm,
    physical_addr_t gpa, physical_addr_t off, void *src, u32 len, bool cacheable);
```

actor 必须来自真实调用上下文；shm 必须是已持有效引用的对象。接口先验证对象范围与地址加法溢出，再检查该 Guest 的 GPA 映射覆盖整个请求、真实 HPA 恰好等于 `shm->addr+off`、类型为真实 RAM/共享区，以及写入不为 readonly。任何拒绝返回 0，并记录对应 actor 的范围或权限事件；不会执行 backing 访问。

第一版接口明确拒绝 alias 映射，避免绕过中间别名权限。产品要支持 alias，需补齐完整权限传播后再放行。调用者保证 Guest/shm 生命周期，并序列化映射配置变更；不能在授权检查和实际访问间撤销映射后释放对象。

原 `vmm_shmem_read/write/set` 仍为宿主内部 API，不猜测当前 vCPU 就是请求者，也不擅自改变所有宿主访问语义。DM/CCM 中由 Guest 发起的共享访问，应改用新接口或在既有 ACL 拒绝点调用 `vmm_monitor_mem_fault()`。上游没有这些私有调用点，本包不能自动替换它们。

对象内部已授权页上的错误访问、不同对象共用同一映射页、恶意 DMA 不会全部经过这些 API。不能把页级 MMU 边界监控宣称成字节级任意指针检测。

## 7. GICv3 / LPI / fastpath 接入

通用配对表按 vCPU 建表，virq 作为存储的 key，**不会使用 LPI ID 索引普通 IRQ 数组**。

```c
vmm_monitor_irq_pending_at(target_vcpu, guest_lpi, original_hv_timestamp);
vmm_monitor_irq_injected(target_vcpu, guest_lpi); /* 成功写 LR 之后 */
vmm_monitor_irq_cancel(target_vcpu, guest_lpi);   /* 取消，不产生成功样本 */
vmm_monitor_irq_move(old_vcpu, new_vcpu, guest_lpi); /* 同域 MOVI，保留原起点 */
```

pending_at 用于异步 DM/CCM 或可显式携带时间戳的 LPI 路径；时间必须为与 HV timer 同源的原始事件时刻，不能在快路径末尾重新取一个“起点”。同步物理分发路径仍可用 `vmm_monitor_irq_pending(target, virq, hirq)` 自动校验来源。

move 只允许同 Guest 的不同 vCPU，按 ID 顺序取得两把关联锁；目标同 virq 已存在时保留较早起点，目标满或跨 epoch 则作废并增加 untracked。调用者必须用实际路由锁同步 MOVI、pending 和注入，否则仅移动监控表不能使真实路由原子化。

针对之前三份附件：

| 产品函数 | 手工接入要求 |
|---|---|
| `__vgic_set_pending()` | 普通 pending 原置位必须保留；LPI 使用实际目标与 LPI 映射建立配对，不访问 `irq_state[lpi]` |
| `__vgic_clear_pending()` | LPI 分支也取消关联 |
| `__vgic_percpu_irq_fastpath()` | 在直接 queue 前补起点；成功 LR 后唯一结束，避免重复记两次 |
| `__vgic_mov_lpi()` | 实际迁移的同一临界区调用 move，不调用会重建起点的普通 pending |
| `vmm_host_generic_irq_exec()` | 成功 lookup 之后 enter；任何 enter 后的返回都恢复 old context |

旧 trace 的 context-switch 执行时间可继续保留，不代替 READY 队列时间。旧 pending 统计若每次覆盖起点，与本版保留首次起点的口径不同，应区分报表。

"""Measure allocated ROM/RAM sections of ELF64 little-endian objects.
Does not count ELF headers, debug info or relocation records as ROM.
Usage: python elf_budget.py --limit 5120 obj1.o obj2.o ...
Exit status 1 means measured objects exceed the budget.
Final linked-image before/after comparison remains the release criterion.
"""
import argparse
import struct
from pathlib import Path

def sections(path):
    data=Path(path).read_bytes()
    if data[:6]!=b'\x7fELF\x02\x01': raise ValueError('Expected ELF64 LE: '+str(path))
    offset=struct.unpack_from('<Q',data,40)[0]
    size,count,names=struct.unpack_from('<HHH',data,58)
    headers=[struct.unpack_from('<IIQQQQIIQQ',data,offset+i*size) for i in range(count)]
    shstr=headers[names]
    strings=data[shstr[4]:shstr[4]+shstr[5]]
    for h in headers:
        name=strings[h[0]:].split(b'\0',1)[0].decode()
        if h[2]&2: yield name, h[5], h[1]==8

def totals(path):
    rom=ram=0
    for name,size,bss in sections(path):
        if bss: ram+=size
        else: rom+=size
    return rom,ram

if __name__=='__main__':
    p=argparse.ArgumentParser()
    p.add_argument('--limit',type=int,default=5120)
    p.add_argument('objects',nargs='+')
    args=p.parse_args()
    rom=ram=0
    for path in args.objects:
        r,b=totals(path); rom+=r; ram+=b
        print(f'{Path(path).name}: ROM={r}, BSS={b}')
    print(f'TOTAL ROM={rom}, BSS={ram}; budget={args.limit}: '+('PASS' if rom<=args.limit else 'FAIL'))
    raise SystemExit(rom>args.limit)

static int number(const char *s, unsigned *value)
{
	char *end; unsigned long n;
	if (!*s || *s == '-') return -1;
	errno = 0; n = strtoul(s, &end, 10);
	if (errno || *end || n > UINT_MAX) return -1;
	*value = (unsigned)n; return 0;
}

#ifndef XM_PARSE_H

struct xm_cpu { uint64_t field[8]; };
static int xm_cpu_read(FILE *f, struct xm_cpu *c)
{
	char line[512];
	return !fgets(line, sizeof(line), f) || sscanf(line,
		"cpu %"SCNu64" %"SCNu64" %"SCNu64" %"SCNu64" %"SCNu64" %"SCNu64" %"SCNu64" %"SCNu64,
		&c->field[0], &c->field[1], &c->field[2], &c->field[3],
		&c->field[4], &c->field[5], &c->field[6], &c->field[7]) != 8 ? -1 : 0;
}
static int xm_cpu_usage(const struct xm_cpu *a, const struct xm_cpu *b, uint64_t *bp)
{
	uint64_t total = 0, idle = 0, delta;
	unsigned i;
	for (i = 0; i < 8; i++) {
		if (b->field[i] < a->field[i]) return -1;
		delta = b->field[i] - a->field[i];
		if (UINT64_MAX - total < delta) return -1;
		total += delta;
		if (i == 3 || i == 4) idle += delta;
	}
	if (!total) return -1;
	*bp = (uint64_t)(((__uint128_t)(total - idle) * 10000) / total);
	return 0;
}
static int xm_mem_read(FILE *f, uint64_t *total, uint64_t *available, uint64_t *free_bytes)
{
	char line[256], key[64], unit[8];
	uint64_t value;
	unsigned present = 0;
	*total = *available = *free_bytes = 0;
	while (fgets(line, sizeof(line), f)) {
		if (sscanf(line, "%63s %"SCNu64" %7s", key, &value, unit) != 3) continue;
		if (strcmp(key, "MemTotal:") && strcmp(key, "MemAvailable:") && strcmp(key, "MemFree:")) continue;
		if (strcmp(unit, "kB") || value > UINT64_MAX / 1024) return -1;
		value *= 1024;
		if (!strcmp(key, "MemTotal:")) { *total = value; present |= 1; }
		if (!strcmp(key, "MemAvailable:")) { *available = value; present |= 2; }
		if (!strcmp(key, "MemFree:")) { *free_bytes = value; present |= 4; }
	}
	return present != 7 || !*total || *available > *total || *free_bytes > *total ? -1 : 0;
}

"""Read pure monitor JSONL with Python integer precision.

Usage: python parse_monitor_jsonl.py qemu_monitor_sample_20261021.jsonl --output analysis.json
This validates exported fields, not actual hardware behavior or full IRQ coverage.
"""
import argparse
import json
from pathlib import Path

NS = 1_000_000_000
QUALITIES = {"disabled", "warmup", "stale", "valid", "warmup_or_gap"}


def unique_object(pairs):
    result = {}
    for key, value in pairs:
        if key in result:
            raise ValueError("duplicate field: " + key)
        result[key] = value
    return result


def uint(record, key):
    value = record[key]
    if type(value) is not int or not 0 <= value <= (1 << 64) - 1:
        raise ValueError(key + " must be an unsigned integer")
    return value


def reject_constant(value):
    raise ValueError("non-JSON number: " + value)


def classify(record):
    if "instruction_cycles" in record:
        return "diagnostic"
    if type(record.get("schema")) is not int or record["schema"] not in (1, 2):
        raise ValueError("unsupported schema")
    if record.get("kind") in {"guest_os", "trap_diagnostic", "diagnostic_summary"}:
        return record["kind"]
    if "seq" in record and "type" in record:
        return "event"
    if "enabled" in record and "samples" in record:
        return "status"
    if "ready" in record and "host_irq_to_lr" in record:
        return "snapshot"
    raise ValueError("unrecognized record type")


def latency(record):
    n = uint(record, "count")
    mean = uint(record, "mean_ns")
    p99 = uint(record, "p99_upper_ns")
    maximum = uint(record, "max_ns")
    jitter = uint(record, "jitter_range_ns")
    if not n:
        if any([mean, p99, maximum, jitter]):
            raise ValueError("zero count has nonzero statistics")
        return dict(count=0, mean_us=None, p99_upper_us=None,
                    max_us=None, min_us=None, jitter_range_us=None)
    if jitter > maximum or not maximum - jitter <= mean <= maximum:
        raise ValueError("latency mean/min/max/jitter are inconsistent")
    # A bucket upper bound can exceed the observed maximum. Do not reject it.
    return dict(count=n, mean_us=mean / 1000, p99_upper_us=p99 / 1000,
                max_us=maximum / 1000, min_us=(maximum - jitter) / 1000,
                jitter_range_us=jitter / 1000)


def analyze(path):
    snapshots, other, errors = [], [], []
    for line_number, line in enumerate(path.read_text(encoding="utf-8-sig").splitlines(), 1):
        if not line.strip():
            continue
        try:
            record = json.loads(line, object_pairs_hook=unique_object,
                                parse_constant=reject_constant)
            if not isinstance(record, dict):
                raise ValueError("record must be an object")
            kind = classify(record)
            if kind != "snapshot":
                other.append(dict(line=line_number, kind=kind, record=record))
                continue
            for key in ["guest", "vcpu", "generation", "hcpu", "state", "time_ns",
                        "interval_ns", "exits", "exit_rate", "last_exit", "ready_age_ns",
                        "faults_pending_summary", "last_address", "last_fault", "last_detail"]:
                uint(record, key)
            if record["quality"] not in QUALITIES:
                raise ValueError("unknown quality")
            dt = record["interval_ns"]
            if record["quality"] == "valid" and not NS * 3 // 4 <= dt <= NS * 5 // 4:
                raise ValueError("valid quality with invalid interval")
            # The first warmup reports interval_ns=0; its rate is not meaningful.
            expected_rate = record["exits"] * NS // dt if dt else None
            if expected_rate is not None and record["exit_rate"] != expected_rate:
                raise ValueError("exit_rate does not match floor(exits * 1e9 / interval_ns)")
            snapshots.append(dict(line=line_number, guest=record["guest"], vcpu=record["vcpu"],
                schema=record["schema"], generation=record["generation"], hcpu=record["hcpu"], state_raw=record["state"],
                quality=record["quality"], time_ns=record["time_ns"], interval_ns=dt,
                exits=record["exits"], exit_rate=record["exit_rate"],
                last_exit_raw=record["last_exit"], ready=latency(record["ready"]),
                host_irq_to_lr=latency(record["host_irq_to_lr"]),
                dispatch=latency(record["dispatch"]) if "dispatch" in record else None))
        except (ValueError, KeyError, TypeError) as error:
            errors.append(dict(line=line_number, error=str(error)))
    return dict(snapshot_count=len(snapshots),
                guest_ids=sorted({r["guest"] for r in snapshots}),
                observed_hcpus=sorted({r["hcpu"] for r in snapshots}),
                irq_sample_count=sum(r["host_irq_to_lr"]["count"] for r in snapshots),
                errors=errors, snapshots=snapshots, other_records=other,
                note="valid denotes window quality, not system health; null latency means no samples")


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("input", type=Path)
    parser.add_argument("--output", type=Path)
    args = parser.parse_args()
    result = analyze(args.input)
    text = json.dumps(result, ensure_ascii=False, indent=2)
    if args.output:
        args.output.write_text(text + "\n", encoding="utf-8")
    else:
        print(text)
    return bool(result["errors"])


if __name__ == "__main__":
    raise SystemExit(main())

	register u64 x6 asm("x6") = r->free_bytes;
	register u64 x7 asm("x7") = r->valid;
	/* The private ABI preserves x1..x7. Declare them read/write conservatively. */
	asm volatile("hvc #0x4d4f" : "+r"(x0), "+r"(x1), "+r"(x2), "+r"(x3),
		"+r"(x4), "+r"(x5), "+r"(x6), "+r"(x7) : : "memory");
	return (long)x0;
}

static ssize_t xm_write(struct file *file, const char __user *buffer,
			size_t length, loff_t *position)
{
	struct xm_report r;
	long rc;
	if (!capable(CAP_SYS_ADMIN)) return -EPERM;
	if (length != sizeof(r)) return -EINVAL;
	if (copy_from_user(&r, buffer, sizeof(r))) return -EFAULT;
	if (!r.valid || r.valid > 3 || r.cpu_bp > 10000 ||
	    r.available_bytes > r.total_bytes || r.free_bytes > r.total_bytes ||
	    ((r.valid & 2) && !r.total_bytes)) return -EINVAL;
	if (mutex_lock_interruptible(&report_lock)) return -ERESTARTSYS;
	rc = xm_call(&r);
	mutex_unlock(&report_lock);
	/* HV error numbers are not Linux errno; do not pass them through. */
	return rc ? -EAGAIN : sizeof(r);
}

static const struct file_operations xm_fops = {
	.owner = THIS_MODULE, .write = xm_write, .open = nonseekable_open,
};
static struct miscdevice xm_device = {
	.minor = MISC_DYNAMIC_MINOR, .name = "xvisor_monitor",
	.fops = &xm_fops, .mode = 0600,
};
static int __init xm_init(void)
{
	get_random_bytes(&session, sizeof(session));
	if (!session) session = 1;
	return misc_register(&xm_device);
}
static void __exit xm_exit(void) { misc_deregister(&xm_device); }
module_init(xm_init);
module_exit(xm_exit);
