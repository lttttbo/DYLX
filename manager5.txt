static struct vmm_vcpu
vcast_orphan_vcpu_array[CONFIG_MAX_VCPU_COUNT];

static bool
vcast_orphan_vcpu_avail_array[CONFIG_MAX_VCPU_COUNT];

static struct vmm_cpumask
vcast_orphan_affinity_array[CONFIG_MAX_VCPU_COUNT];

static u32
vcast_orphan_domain_id_array[CONFIG_MAX_VCPU_COUNT];

/*
 * 改写目的：
 * 为 orphan_create 准备 manager 内部 VCPU 数组、可用状态数组、
 * affinity 数组、domain ID 数组和孤儿 VCPU 链表。
 *
 * first_available_index：
 *   -1  -> 没有可用 VCPU；
 *   0及以上有效索引 -> 对应 VCPU 可用。
 *
 * 返回值：
 *   对应的可用 VCPU 地址；没有可用 VCPU 时返回 NULL。
 */
struct vmm_vcpu *
vcast_manager_prepare_orphan_create_env(int first_available_index)
{
    u32 i;

    memset(vcast_orphan_vcpu_array,
           0,
           sizeof(vcast_orphan_vcpu_array));

    memset(vcast_orphan_vcpu_avail_array,
           0,
           sizeof(vcast_orphan_vcpu_avail_array));

    memset(vcast_orphan_affinity_array,
           0,
           sizeof(vcast_orphan_affinity_array));

    memset(vcast_orphan_domain_id_array,
           0,
           sizeof(vcast_orphan_domain_id_array));

    mngr.vcpu_array = vcast_orphan_vcpu_array;
    mngr.vcpu_avail_array = vcast_orphan_vcpu_avail_array;
    mngr.vcpu_affinity_mask = vcast_orphan_affinity_array;
    mngr.vcpu_hcpu_domain_id = vcast_orphan_domain_id_array;
    mngr.vcpu_count = 0U;

    INIT_LIST_HEAD(&mngr.orphan_vcpu_list);

    for (i = 0U; i < (u32)CONFIG_MAX_VCPU_COUNT; ++i) {
        vcast_orphan_vcpu_array[i].id = i;
        vcast_orphan_vcpu_avail_array[i] = FALSE;
        INIT_LIST_HEAD(&vcast_orphan_vcpu_array[i].head);
    }

    if ((first_available_index >= 0) &&
        ((u32)first_available_index <
            (u32)CONFIG_MAX_VCPU_COUNT)) {
        vcast_orphan_vcpu_avail_array[
            (u32)first_available_index] = TRUE;

        return &vcast_orphan_vcpu_array[
            (u32)first_available_index];
    }

    return NULL;
}

/*
 * 改写目的：
 * 为 orphan_destroy 准备可用状态数组、孤儿 VCPU 链表和计数。
 * 传入的 VCPU 被视为当前已创建、尚未释放的孤儿 VCPU。
 */
void vcast_manager_prepare_orphan_destroy_env(
    struct vmm_vcpu *vcpu)
{
    u32 i;

    memset(vcast_orphan_vcpu_avail_array,
           0,
           sizeof(vcast_orphan_vcpu_avail_array));

    memset(vcast_orphan_affinity_array,
           0,
           sizeof(vcast_orphan_affinity_array));

    memset(vcast_orphan_domain_id_array,
           0,
           sizeof(vcast_orphan_domain_id_array));

    mngr.vcpu_avail_array = vcast_orphan_vcpu_avail_array;
    mngr.vcpu_affinity_mask = vcast_orphan_affinity_array;
    mngr.vcpu_hcpu_domain_id = vcast_orphan_domain_id_array;
    mngr.vcpu_count = 1U;

    INIT_LIST_HEAD(&mngr.orphan_vcpu_list);

    for (i = 0U; i < (u32)CONFIG_MAX_VCPU_COUNT; ++i) {
        vcast_orphan_vcpu_avail_array[i] = TRUE;
    }

    if (!vcpu ||
        ((u32)vcpu->id >=
            (u32)CONFIG_MAX_VCPU_COUNT)) {
        return;
    }

    vcast_orphan_vcpu_avail_array[vcpu->id] = FALSE;

    INIT_LIST_HEAD(&vcpu->head);
    list_add_tail(&vcpu->head, &mngr.orphan_vcpu_list);
}
